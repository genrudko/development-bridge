def run():
    op = PAYLOAD.get("operation", "")
    if op == "test_op":
        return make_result(data={"operation": op}, summary=f"Executed mutate:{op}")

    _RESERVED_GROUP = "bridge.cad/v1"
    _PROVENANCE_NAME = "provenance"
    _MUTATION_OPS = ("set", "remove", "tag", "untag", "set_role", "clear_role")
    _READ_OPS = ("get", "query", "provenance")

    def _open_runtime():
        import re as _re

        import adsk.core

        app = adsk.core.Application.get()
        doc = getattr(app, "activeDocument", None) if app else None
        design = None
        if doc is not None and hasattr(doc, "products"):
            try:
                design = doc.products.itemByClass("adsk::fusion::Design")
            except Exception:
                design = None
        if not doc or not design:
            raise FusionScriptError(
                "NO_ACTIVE_DESIGN",
                "Active document and design required for metadata operation",
                details={"has_doc": bool(doc), "has_design": bool(design)},
            )
        doc_id = (
            getattr(doc, "dataId", None)
            or getattr(doc, "creationId", None)
            or getattr(getattr(doc, "dataFile", None), "id", None)
        )
        if not doc_id or not str(doc_id).strip():
            raise FusionScriptError(
                "NO_ACTIVE_DESIGN",
                "Document lacks stable runtime identity; mutable document name cannot be used as authoritative identity",
                details={"name": getattr(doc, "name", None)},
            )
        clean = _re.sub(r"[^A-Za-z0-9._-]", "_", str(doc_id).strip())
        safe_doc = clean if clean.startswith("doc_") else f"doc_{clean}"
        req_doc = PAYLOAD.get("document_ref")
        if req_doc and req_doc != safe_doc:
            raise FusionScriptError(
                "WRONG_DOCUMENT",
                f"Requested document_ref '{req_doc}' does not match active document runtime identity '{safe_doc}'",
                details={"requested": req_doc, "active": safe_doc},
            )
        return app, doc, design, safe_doc

    def _doc_state(safe_doc, doc):
        return {
            "document_ref": safe_doc,
            "model_revision": PAYLOAD.get("expected_revision") or "rev_1",
            "name": getattr(doc, "name", "active"),
            "units": "mm",
        }

    def _owner_attributes(owner):
        attrs = getattr(owner, "attributes", None)
        if attrs is None or not hasattr(attrs, "count") or not hasattr(attrs, "item"):
            raise FusionScriptError(
                "CAPABILITY_UNAVAILABLE",
                "Owner attribute collection is unreadable; metadata cannot be accessed safely",
                details={},
            )
        items = []
        for idx in range(attrs.count):
            a = attrs.item(idx)
            if a is None:
                raise FusionScriptError(
                    "FUSION_API_ERROR",
                    f"Attribute {idx} on metadata owner is unreadable (None)",
                )
            items.append(a)
        return attrs, items

    def _read_records(owner, owner_type, group_filter=None, name_filter=None, value_filter=None):
        _, items = _owner_attributes(owner)
        records = []
        for a in items:
            grp = str(getattr(a, "groupName", "") or "")
            nm = str(getattr(a, "name", "") or "")
            if group_filter and grp != group_filter:
                continue
            if name_filter and nm != name_filter:
                continue
            val = str(getattr(a, "value", "") or "")
            if value_filter is not None and val != value_filter:
                continue
            records.append(
                {"group": grp, "name": nm, "value": val, "owner_type": owner_type}
            )
        return records

    def _resolve_owner(doc, design, target):
        """Resolve the metadata owner EXACTLY; fail closed, never document fallback.

        - No target at all: explicit document-scope metadata (document owner).
        - Mapping target with a native token: exact entity-token resolution;
          ambiguous or stale resolutions fail closed (REF_AMBIGUOUS/REF_STALE).
        - Mapping target WITHOUT a native token and any raw string ref: fail
          closed with REF_STALE. An unresolvable entity target is never
          silently mapped onto the document owner, so document metadata is
          never written while echoing an entity ref.
        """
        if target is None:
            return doc, None
        if isinstance(target, dict):
            token = target.get("native_token")
            if not token:
                raise FusionScriptError(
                    "REF_STALE",
                    "Metadata target hint lacks a native token and cannot be resolved exactly; refusing document fallback",
                    details={"ref": target.get("ref")},
                )
            if not hasattr(design, "findEntityByToken"):
                raise FusionScriptError(
                    "CAPABILITY_UNAVAILABLE",
                    "Entity-token resolver unavailable; cannot resolve metadata target exactly",
                    details={"ref": target.get("ref")},
                )
            try:
                resolved = design.findEntityByToken(token)
            except Exception:
                resolved = None
            matches = []
            if resolved is not None:
                if hasattr(resolved, "count"):
                    matches = [resolved.item(i) for i in range(resolved.count)]
                else:
                    matches = [resolved]
            if len(matches) == 1:
                return matches[0], target.get("ref")
            if len(matches) > 1:
                raise FusionScriptError(
                    "REF_AMBIGUOUS",
                    "Metadata target resolves to multiple current entities; refusing to pick one",
                    details={"ref": target.get("ref")},
                )
            raise FusionScriptError(
                "REF_STALE",
                "Metadata target no longer resolves to a current entity",
                details={"ref": target.get("ref")},
            )
        raise FusionScriptError(
            "REF_STALE",
            "Metadata target is an unresolved opaque entity ref; unknown or stale refs fail closed and never fall back to the document owner",
            details={"target_type": type(target).__name__},
        )

    def _public_target(target, target_ref):
        if target_ref:
            return target_ref
        if isinstance(target, str):
            return target
        return None

    def _reserved_records(owner):
        """Snapshot of the owner's reserved-group attributes as (name, value) pairs."""
        _attrs, items = _owner_attributes(owner)
        records = []
        for a in items:
            if str(getattr(a, "groupName", "") or "") != _RESERVED_GROUP:
                continue
            records.append(
                (
                    str(getattr(a, "name", "") or ""),
                    str(getattr(a, "value", "") or ""),
                )
            )
        return records

    def _preflight_metadata_plan(owner, writes, removals):
        """Verify every capability the metadata plan needs BEFORE any mutation.

        Fails closed (CAPABILITY_UNAVAILABLE) when attribute writes/removals or
        the compensation primitives (Attribute.deleteMe / Attributes.add) are
        unavailable, so a plan that could not be rolled back never starts
        applying. Read-only: applies nothing.

        Corrupted-state preflight: Autodesk documents attribute names as unique
        within a group, so duplicate same-(group, name) records inside the
        reserved namespace are outside the normal API contract. Their multiset
        cannot be reconstructed by Attributes.add, so the plan is rejected
        BEFORE the geometry hook or any reserved attribute side effect instead
        of attempting impossible compensation. Unrelated groups are untouched.
        """
        attrs, items = _owner_attributes(owner)
        if writes and not hasattr(attrs, "add"):
            raise FusionScriptError(
                "CAPABILITY_UNAVAILABLE",
                "Attribute collection does not support writes; failing closed before any mutation",
                details={"operation": op},
            )
        seen_reserved_names = set()
        for a in items:
            if str(getattr(a, "groupName", "") or "") != _RESERVED_GROUP:
                continue
            nm = str(getattr(a, "name", "") or "")
            if nm in seen_reserved_names:
                raise FusionScriptError(
                    "CAPABILITY_UNAVAILABLE",
                    "Owner reserved metadata contains duplicate attribute names; the corrupted state cannot be safely mutated or compensated, so the plan fails closed before any mutation",
                    details={"operation": op},
                )
            seen_reserved_names.add(nm)
        removal_requests = []
        for removal in removals:
            if not isinstance(removal, dict):
                raise FusionScriptError(
                    "INVALID_ARGUMENT",
                    "Metadata removal plan contains a non-object entry",
                    details={"operation": op},
                )
            rm_name = str(removal.get("name", "") or "")
            if not rm_name:
                raise FusionScriptError(
                    "INVALID_ARGUMENT",
                    "Metadata removal plan contains an empty attribute name",
                    details={"operation": op},
                )
            removal_requests.append((rm_name, removal.get("value")))
        for a in items:
            if str(getattr(a, "groupName", "") or "") != _RESERVED_GROUP:
                continue
            nm = str(getattr(a, "name", "") or "")
            val = str(getattr(a, "value", "") or "")
            for rm_name, expected_val in removal_requests:
                if nm != rm_name:
                    continue
                if expected_val is not None and val != str(expected_val):
                    continue
                if not hasattr(a, "deleteMe"):
                    raise FusionScriptError(
                        "CAPABILITY_UNAVAILABLE",
                        "Attribute does not support deleteMe; failing closed instead of leaving stale Bridge metadata",
                        details={"operation": op},
                    )

    def _rollback_delete_reserved(owner, delete_targets, name, value=None):
        """Delete reserved attribute(s) during compensation with full proof.

        Documented Autodesk semantics: Attribute.deleteMe returns True iff the
        deletion succeeded. A misbehaving runtime may also report success
        without removing anything, so the owner is re-enumerated after the
        deletes and the targeted reserved record (name, or name+value when the
        value is known) must be observably ABSENT. Either failure raises
        FUSION_API_ERROR with compensated=False: a bad Boolean result that
        nevertheless mutated state is still a compensation failure, and success
        is never inferred from the final snapshot equality alone. Only safe
        constant metadata (operation/attribute_name/compensated) is attached to
        the error; the raw deleteMe result object is never stringified or
        copied into details."""
        for delete_attr in delete_targets:
            if not hasattr(delete_attr, "deleteMe"):
                raise FusionScriptError(
                    "CAPABILITY_UNAVAILABLE",
                    "Attribute does not support deleteMe; metadata compensation is unavailable and partial metadata may remain",
                    details={
                        "operation": op,
                        "attribute_name": name,
                        "compensated": False,
                    },
                )
            delete_result = delete_attr.deleteMe()
            if delete_result is not True:
                raise FusionScriptError(
                    "FUSION_API_ERROR",
                    "Compensation Attribute.deleteMe did not report success; metadata compensation is unavailable and partial metadata may remain",
                    details={
                        "operation": op,
                        "attribute_name": name,
                        "compensated": False,
                    },
                )
        _verify_attrs, verify_items = _owner_attributes(owner)
        for post_attr in verify_items:
            if str(getattr(post_attr, "groupName", "") or "") != _RESERVED_GROUP:
                continue
            if str(getattr(post_attr, "name", "") or "") != name:
                continue
            if value is not None and str(
                getattr(post_attr, "value", "") or ""
            ) != str(value):
                continue
            raise FusionScriptError(
                "FUSION_API_ERROR",
                "Compensation Attribute.deleteMe reported success but the targeted reserved attribute is still present; metadata compensation is unavailable and partial metadata may remain",
                details={
                    "operation": op,
                    "attribute_name": name,
                    "compensated": False,
                },
            )

    def _rollback_metadata(owner, undo):
        """Restore the exact pre-mutation reserved-namespace state (all-or-nothing).

        Uses ONLY the verified primitives this script already uses for metadata
        (Attribute.deleteMe and Attributes.add); no transaction/undo APIs are
        invented. The undo log distinguishes attributes this plan CREATED
        (undone via deleteMe) from pre-existing attributes this plan OVERWROTE
        or REMOVED (undone by re-adding the captured old value): real Fusion
        Attributes.add(groupName, name, value) updates and returns the existing
        same-group/same-name attribute instead of creating a new one, so a
        returned Attribute is NOT proof of creation and deleteMe-ing it would
        destroy pre-existing provenance/metadata instead of restoring its old
        value. A failed CURRENT write whose Attributes.add returned null or a
        mismatched Attribute is compensated through ("restored", name,
        old_value) when it overwrote a pre-existing attribute, or through
        ("delete_created_by_name", name) when it may have created one; the
        persisted state is re-derived by enumeration and the bad returned
        object is never trusted as proof of what was created. Every rollback
        delete (created attribute or re-derived failed-write record) must
        return the documented True deleteMe result AND re-enumeration must
        prove the targeted reserved record is observably absent; a False/bad
        result — even one that mutated state — fails the compensation closed.
        Raises when the restored state does not match the pre-mutation snapshot
        so a partially applied plan can never be reported as success.
        """
        if undo is None:
            return
        undo_actions, snapshot = undo
        restore_attrs = getattr(owner, "attributes", None)
        for action in reversed(undo_actions):
            kind = action[0]
            if kind == "created":
                # The undo entry captures the created attribute together with
                # its verified reserved name/value (captured when the write was
                # validated, never re-read from the possibly misbehaving
                # runtime object later). The compensation delete must report
                # the documented True result AND the created record must be
                # observably absent afterwards.
                attr, created_name, created_value = action[1], action[2], action[3]
                _rollback_delete_reserved(
                    owner, [attr], created_name, created_value
                )
            elif kind == "restored":
                name, value = action[1], action[2]
                if restore_attrs is None or not hasattr(restore_attrs, "add"):
                    raise FusionScriptError(
                        "CAPABILITY_UNAVAILABLE",
                        "Attribute collection does not support writes; metadata compensation is unavailable and partial metadata may remain",
                        details={"operation": op, "compensated": False},
                    )
                restored_attr = restore_attrs.add(_RESERVED_GROUP, name, value)
                if restored_attr is None:
                    raise FusionScriptError(
                        "FUSION_API_ERROR",
                        "Compensation Attributes.add returned null; metadata compensation is unavailable and partial metadata may remain",
                        details={"operation": op, "compensated": False},
                    )
                restored_group = str(getattr(restored_attr, "groupName", "") or "")
                restored_name = str(getattr(restored_attr, "name", "") or "")
                restored_value = str(getattr(restored_attr, "value", "") or "")
                if (
                    restored_group != _RESERVED_GROUP
                    or restored_name != name
                    or restored_value != value
                ):
                    raise FusionScriptError(
                        "FUSION_API_ERROR",
                        "Compensation Attributes.add returned an Attribute that does not match the restore write; metadata compensation is unavailable and partial metadata may remain",
                        details={"operation": op, "compensated": False},
                    )
            elif kind == "delete_created_by_name":
                # A failed/untrusted Attributes.add result can still have
                # persisted created state under this reserved name. Re-derive
                # the actual persisted attributes by enumeration and delete
                # exactly what the failed write created; the null/mismatched
                # result object is never trusted as proof of creation. Every
                # delete must report the documented True result AND matching
                # reserved name(s) must be observably absent afterwards.
                name = action[1]
                _del_attrs, del_items = _owner_attributes(owner)
                delete_targets = [
                    del_attr
                    for del_attr in del_items
                    if str(getattr(del_attr, "groupName", "") or "")
                    == _RESERVED_GROUP
                    and str(getattr(del_attr, "name", "") or "") == name
                ]
                _rollback_delete_reserved(owner, delete_targets, name)
            else:
                raise FusionScriptError(
                    "FUSION_API_ERROR",
                    "Unknown undo action; metadata compensation is unavailable and partial metadata may remain",
                    details={"operation": op, "compensated": False},
                )
        current = sorted(_reserved_records(owner))
        if current != sorted(snapshot):
            raise FusionScriptError(
                "FUSION_API_ERROR",
                "Metadata compensation did not restore the pre-mutation attribute state; partial metadata may remain",
                details={"operation": op, "compensated": False},
            )

    def _apply_metadata_plan(owner, writes, removals):
        """Apply the plan with an undo log; any failure compensates fully.

        Returns (applied_writes, applied_removals, undo). On any application
        failure the reserved-namespace state is restored to the exact
        pre-mutation snapshot before the failure is re-raised, so a failed
        command never leaves partial metadata/provenance behind. Each write
        captures its pre-add reserved state BEFORE calling Attributes.add, so a
        failed current write with a null/mismatched add result is itself
        compensated (restored by re-adding the old value, or deleted by
        enumerated name) instead of remaining applied while only earlier
        actions are rolled back.
        """
        snapshot = _reserved_records(owner)
        undo_actions = []
        undo = (undo_actions, snapshot)
        applied_writes = []
        applied_removals = []
        try:
            for removal in removals:
                if not isinstance(removal, dict):
                    raise FusionScriptError(
                        "INVALID_ARGUMENT",
                        "Metadata removal plan contains a non-object entry",
                        details={"operation": op},
                    )
                rm_name = str(removal.get("name", "") or "")
                if not rm_name:
                    raise FusionScriptError(
                        "INVALID_ARGUMENT",
                        "Metadata removal plan contains an empty attribute name",
                        details={"operation": op},
                    )
                expected_val = removal.get("value")
                _attrs, items = _owner_attributes(owner)
                for a in items:
                    if str(getattr(a, "groupName", "") or "") != _RESERVED_GROUP:
                        continue
                    if str(getattr(a, "name", "") or "") != rm_name:
                        continue
                    if expected_val is not None and str(
                        getattr(a, "value", "") or ""
                    ) != str(expected_val):
                        continue
                    if not hasattr(a, "deleteMe"):
                        raise FusionScriptError(
                            "CAPABILITY_UNAVAILABLE",
                            "Attribute does not support deleteMe; failing closed instead of leaving stale Bridge metadata",
                            details={"operation": op},
                        )
                    removed_value = str(getattr(a, "value", "") or "")
                    # Register the CURRENT removal undo BEFORE deleting so a
                    # deleteMe that mutates persisted state but reports a bad
                    # result is still compensable back to the exact record.
                    undo_actions.append(("restored", rm_name, removed_value))
                    # Documented Autodesk semantics: Attribute.deleteMe returns
                    # True iff the deletion succeeded; False means the attribute
                    # was NOT removed. The return value alone is never trusted
                    # (a misbehaving runtime may report success without removing
                    # anything), so the owner is re-enumerated and the exact
                    # targeted reserved (group, name, value) record must have
                    # disappeared before the removal is treated as applied.
                    # False/no-op/mismatch fails closed; the except handler then
                    # restores the exact pre-mutation snapshot.
                    delete_result = a.deleteMe()
                    if delete_result is not True:
                        raise FusionScriptError(
                            "FUSION_API_ERROR",
                            "Attribute.deleteMe did not report success for the metadata removal; failing closed so stale metadata is never reported as applied",
                            # Confidentiality: only safe constant metadata is
                            # attached; the arbitrary raw deleteMe result
                            # object is never stringified or copied here.
                            details={
                                "operation": op,
                                "attribute_name": rm_name,
                            },
                        )
                    _verify_attrs, verify_items = _owner_attributes(owner)
                    for post_attr in verify_items:
                        post_group = str(getattr(post_attr, "groupName", "") or "")
                        if post_group != _RESERVED_GROUP:
                            continue
                        post_name = str(getattr(post_attr, "name", "") or "")
                        if post_name != rm_name:
                            continue
                        post_value = str(getattr(post_attr, "value", "") or "")
                        if post_value != removed_value:
                            continue
                        raise FusionScriptError(
                            "FUSION_API_ERROR",
                            "Attribute.deleteMe reported success but the targeted reserved attribute is still present; failing closed so stale metadata is never reported as applied",
                            details={"operation": op, "attribute_name": rm_name},
                        )
                    applied_removals.append(
                        {
                            "name": rm_name,
                            "value": str(expected_val)
                            if expected_val is not None
                            else None,
                        }
                    )

            for write in writes:
                if not isinstance(write, dict):
                    raise FusionScriptError(
                        "INVALID_ARGUMENT",
                        "Metadata write plan contains a non-object entry",
                        details={"operation": op},
                    )
                wr_name = str(write.get("name", "") or "")
                wr_value = write.get("value")
                if not wr_name or wr_value is None:
                    raise FusionScriptError(
                        "INVALID_ARGUMENT",
                        "Metadata write plan contains an empty attribute name or missing value",
                        details={"operation": op},
                    )
                attrs, items = _owner_attributes(owner)
                if not hasattr(attrs, "add"):
                    raise FusionScriptError(
                        "CAPABILITY_UNAVAILABLE",
                        "Attribute collection does not support writes; failing closed",
                        details={"operation": op},
                    )
                # Real Fusion semantics: Attributes.add UPDATES and returns the
                # EXISTING same-group/same-name attribute when the owner already
                # has one. Capture the pre-add state so the undo log can
                # distinguish a newly created attribute (undone via deleteMe)
                # from an overwritten pre-existing attribute (undone by
                # re-adding its captured old value).
                existing_value = None
                existed = False
                for a in items:
                    if str(getattr(a, "groupName", "") or "") != _RESERVED_GROUP:
                        continue
                    if str(getattr(a, "name", "") or "") != wr_name:
                        continue
                    existed = True
                    existing_value = str(getattr(a, "value", "") or "")
                    break
                new_attr = attrs.add(_RESERVED_GROUP, wr_name, str(wr_value))
                # Verified Fusion semantics: Attributes.add(groupName, name,
                # value) returns the created or updated Attribute, or null when
                # creation fails. The returned Attribute is validated BEFORE the
                # write is marked applied or enters the undo log, so a null or
                # mismatched result can never be reported as an applied write
                # (nor deleted as a fake "created" attribute). A misbehaving
                # runtime may ALREADY have mutated/created persisted state
                # before returning null/mismatch, so the failed CURRENT write
                # itself is registered for compensation from the pre-add state
                # captured above: an overwritten attribute is restored via
                # Attributes.add(group, name, old_value); a possibly created one
                # is re-derived by enumeration and deleted via
                # Attribute.deleteMe ("delete_created_by_name"). The bad
                # returned object is never trusted as proof of creation. The
                # except handler below then compensates every PRIOR plan
                # write/removal via the undo log, and the post-rollback exact
                # snapshot check in _rollback_metadata fails closed
                # (compensated=False) if exact restoration cannot be proven.
                if new_attr is None:
                    undo_actions.append(
                        ("restored", wr_name, existing_value)
                        if existed
                        else ("delete_created_by_name", wr_name)
                    )
                    raise FusionScriptError(
                        "FUSION_API_ERROR",
                        "Attributes.add returned null for a metadata write; the write cannot be verified as applied",
                        details={"operation": op, "attribute_name": wr_name},
                    )
                added_group = str(getattr(new_attr, "groupName", "") or "")
                added_name = str(getattr(new_attr, "name", "") or "")
                added_value = str(getattr(new_attr, "value", "") or "")
                if (
                    added_group != _RESERVED_GROUP
                    or added_name != wr_name
                    or added_value != str(wr_value)
                ):
                    undo_actions.append(
                        ("restored", wr_name, existing_value)
                        if existed
                        else ("delete_created_by_name", wr_name)
                    )
                    raise FusionScriptError(
                        "FUSION_API_ERROR",
                        "Attributes.add returned an Attribute that does not match the requested metadata write; the write cannot be verified as applied",
                        details={"operation": op, "attribute_name": wr_name},
                    )
                if existed:
                    undo_actions.append(("restored", wr_name, existing_value))
                else:
                    undo_actions.append(
                        ("created", new_attr, wr_name, str(wr_value))
                    )
                applied_writes.append({"name": wr_name, "value": str(wr_value)})
        except FusionScriptError:
            _rollback_metadata(owner, undo)
            raise
        except Exception:
            _rollback_metadata(owner, undo)
            raise FusionScriptError(
                "FUSION_API_ERROR",
                "Metadata plan application failed; compensated back to the exact pre-mutation metadata state",
                details={"operation": op, "compensated": True},
            )
        return applied_writes, applied_removals, undo

    def _prepare_geometry_compensation(primitive_hook):
        """Capture operation-specific rollback state before geometry runs.

        These hooks are internal script-scope closures supplied by the trusted
        operation implementation. They are not payload fields or public API.
        """
        if not callable(primitive_hook):
            return None
        capture_hook = globals().get("_mutation_compensation_capture")
        rollback_hook = globals().get("_mutation_compensation_rollback")
        if not callable(capture_hook) or not callable(rollback_hook):
            raise FusionScriptError(
                "CAPABILITY_UNAVAILABLE",
                "Plan-bearing geometry requires an explicit operation-specific compensation contract",
                details={"operation": op, "applied": False, "compensated": False},
            )
        try:
            return capture_hook(PAYLOAD)
        except Exception:
            raise FusionScriptError(
                "CAPABILITY_UNAVAILABLE",
                "Geometry compensation state could not be captured before mutation",
                details={"operation": op, "applied": False, "compensated": False},
            )

    def _compensate_geometry(captured, baseline_fp):
        """Rollback geometry and prove exact restoration by full fingerprint."""
        rollback_hook = globals().get("_mutation_compensation_rollback")
        try:
            rollback_result = rollback_hook(captured)
            if rollback_result is not True:
                raise RuntimeError("rollback did not explicitly report success")
            restored_fp, _restored_canonical, _restored_doc = collect_model_fingerprint(
                PAYLOAD
            )
            if restored_fp != baseline_fp:
                raise RuntimeError("rollback fingerprint does not match baseline")
        except Exception:
            raise FusionScriptError(
                "CAPABILITY_UNAVAILABLE",
                "Geometry compensation failed or exact model restoration could not be verified",
                details={"operation": op, "applied": False, "compensated": False},
            )

    if op in _MUTATION_OPS:
        # Authoritative Fusion-side freshness guard runs in this SAME command
        # execution, with no return to UI between guard and apply.
        expected_fp = PAYLOAD.get("expected_fingerprint")
        current_fp, _canonical, doc_ref = collect_model_fingerprint(PAYLOAD)
        if expected_fp is None or expected_fp != current_fp:
            return make_error(
                "REVISION_CONFLICT",
                "Authoritative Fusion-side guard: model fingerprint diverged from baseline",
                details={
                    "document_ref": doc_ref,
                    "expected_revision": PAYLOAD.get("expected_revision"),
                    "expected_fingerprint": expected_fp,
                    "current_fingerprint": current_fp,
                    "applied": False,
                },
            )

        # The complete metadata side of this ONE command is planned by the Bridge:
        # explicit requested writes/removals PLUS the provenance write. Geometry or
        # metadata change and provenance are applied by this single execution; no
        # hidden post-commit metadata command exists.
        writes = PAYLOAD.get("metadata_writes")
        removals = PAYLOAD.get("metadata_removals")
        if not isinstance(writes, list) or not isinstance(removals, list):
            raise FusionScriptError(
                "INVALID_ARGUMENT",
                "Metadata mutation payload lacks the one-command metadata plan (metadata_writes/metadata_removals)",
                details={"operation": op},
            )

        _app, doc, design, safe_doc = _open_runtime()
        owner, target_ref = _resolve_owner(doc, design, PAYLOAD.get("target"))

        # Preflight verifies every capability the plan needs (including the
        # compensation primitives) BEFORE anything is mutated, and rejects
        # corrupted duplicate reserved (group, name) state before the geometry
        # hook or any reserved attribute side effect.
        _preflight_metadata_plan(owner, writes, removals)

        # Geometry primitive (when present) runs only after operation-specific
        # rollback state is captured. Every later failure compensates metadata
        # and geometry, then proves exact restoration by full fingerprint.
        primitive_hook = globals().get("_mutation_primitive")
        geometry_capture = _prepare_geometry_compensation(primitive_hook)
        geometry_applied = False
        try:
            if callable(primitive_hook):
                geometry_applied = True
                primitive_hook(PAYLOAD)

            applied_writes, applied_removals, undo = _apply_metadata_plan(
                owner, writes, removals
            )
            if not applied_writes:
                _rollback_metadata(owner, undo)
                raise FusionScriptError(
                    "INVALID_ARGUMENT",
                    "Metadata mutation plan produced no attribute writes; provenance cannot be recorded transactionally",
                    details={"operation": op},
                )

        # Freshly collect real post-apply fingerprint of actual model state. Any
        # failure here is compensated so partial metadata is never left behind
        # and the command is never externally reported as applied.
            post_fp, _post_canonical, post_doc_ref = collect_model_fingerprint(PAYLOAD)
        except Exception as exc:
            try:
                if "undo" in locals():
                    _rollback_metadata(owner, undo)
            finally:
                if geometry_applied:
                    _compensate_geometry(geometry_capture, current_fp)
            if isinstance(exc, FusionScriptError):
                raise exc
            raise FusionScriptError(
                "FUSION_API_ERROR",
                "Mutation execution failed after an apply attempt; exact compensation completed",
                details={"operation": op, "applied": False, "compensated": True},
            )
        if not post_fp or not isinstance(post_fp, str) or not post_fp.strip():
            try:
                _rollback_metadata(owner, undo)
            finally:
                if geometry_applied:
                    _compensate_geometry(geometry_capture, current_fp)
            return make_error(
                "FINGERPRINT_FAILED",
                "Authoritative post-apply model fingerprint collection failed or returned empty",
                details={
                    "document_ref": post_doc_ref or doc_ref,
                    "applied": False,
                    "compensated": True,
                },
            )
        if post_fp == current_fp:
            try:
                _rollback_metadata(owner, undo)
            finally:
                if geometry_applied:
                    _compensate_geometry(geometry_capture, current_fp)
            return make_error(
                "MUTATION_FAILED",
                "Mutation failed to produce distinct post-apply model state; cannot advance model revision",
                details={
                    "document_ref": post_doc_ref or doc_ref,
                    "initial_fingerprint": current_fp,
                    "post_fingerprint": post_fp,
                    "applied": False,
                    "compensated": True,
                },
            )

        try:
            result_data = {
                "operation": op,
                "applied": True,
                "fingerprint": post_fp,
                "document_ref": post_doc_ref or doc_ref,
                "target": _public_target(PAYLOAD.get("target"), target_ref),
                "attributes": applied_writes,
                "removed": applied_removals,
                "provenance": PAYLOAD.get("provenance"),
            }
            return make_result(
                data=result_data,
                summary=f"Executed mutate:{op}",
                document=_doc_state(safe_doc, doc),
                changed_refs=[target_ref] if target_ref else [],
            )
        except Exception as exc:
            try:
                _rollback_metadata(owner, undo)
            finally:
                if geometry_applied:
                    _compensate_geometry(geometry_capture, current_fp)
            if isinstance(exc, FusionScriptError):
                raise exc
            raise FusionScriptError(
                "FUSION_API_ERROR",
                "Mutation execution failed after an apply attempt; exact compensation completed",
                details={"operation": op, "applied": False, "compensated": True},
            )

    if op in _READ_OPS:
        # Read-only metadata: no expected_revision, no mutation, no fingerprint write.
        _app, doc, design, safe_doc = _open_runtime()
        group_filter = PAYLOAD.get("group") or _RESERVED_GROUP
        name_filter = PAYLOAD.get("name")
        value_filter = PAYLOAD.get("value")
        if value_filter is not None:
            value_filter = str(value_filter)

        if op == "get":
            owner, target_ref = _resolve_owner(doc, design, PAYLOAD.get("target"))
            records = _read_records(
                owner, "document" if owner is doc else "entity", group_filter, name_filter
            )
            return make_result(
                data={
                    "target": _public_target(PAYLOAD.get("target"), target_ref),
                    "group": group_filter,
                    "records": records,
                    "count": len(records),
                },
                summary=f"Executed mutate:{op}",
                document=_doc_state(safe_doc, doc),
            )

        if op == "provenance":
            owner, target_ref = _resolve_owner(doc, design, PAYLOAD.get("target"))
            records = _read_records(
                owner,
                "document" if owner is doc else "entity",
                _RESERVED_GROUP,
                _PROVENANCE_NAME,
            )
            return make_result(
                data={
                    "target": _public_target(PAYLOAD.get("target"), target_ref),
                    "group": _RESERVED_GROUP,
                    "records": records,
                    "count": len(records),
                },
                summary=f"Executed mutate:{op}",
                document=_doc_state(safe_doc, doc),
            )

        # query: scan all metadata owners and surface records plus selector-ready
        # candidates carrying their persisted model attributes.
        records = []
        candidates = []

        def _collect(owner, owner_type, name_hint, token):
            owner_records = _read_records(
                owner, owner_type, group_filter, name_filter, value_filter
            )
            records.extend(owner_records)
            if owner_records and token:
                candidates.append(
                    {
                        "entityToken": token,
                        "name": name_hint,
                        "kind": owner_type,
                        "attributes": owner_records,
                    }
                )

        def _owner_token(owner):
            token = getattr(owner, "entityToken", None) or getattr(owner, "id", None)
            return str(token).strip() if token else None

        _collect(doc, "document", getattr(doc, "name", None), None)

        comps = getattr(design, "allComponents", None)
        if comps is not None and hasattr(comps, "count"):
            for i in range(comps.count):
                c = comps.item(i)
                if c is not None:
                    _collect(c, "component", getattr(c, "name", None), _owner_token(c))

        root = getattr(design, "rootComponent", None)
        if root is not None:
            occs = getattr(root, "allOccurrences", None)
            if occs is not None and hasattr(occs, "count"):
                for i in range(occs.count):
                    o = occs.item(i)
                    if o is not None:
                        _collect(
                            o, "occurrence", getattr(o, "name", None), _owner_token(o)
                        )

        if comps is not None and hasattr(comps, "count"):
            for i in range(comps.count):
                c = comps.item(i)
                if c is None:
                    continue
                bodies = getattr(c, "bRepBodies", None)
                if bodies is not None and hasattr(bodies, "count"):
                    for bi in range(bodies.count):
                        b = bodies.item(bi)
                        if b is not None:
                            _collect(
                                b, "body", getattr(b, "name", None), _owner_token(b)
                            )
                sketches = getattr(c, "sketches", None)
                if sketches is not None and hasattr(sketches, "count"):
                    for si in range(sketches.count):
                        s = sketches.item(si)
                        if s is not None:
                            _collect(
                                s, "sketch", getattr(s, "name", None), _owner_token(s)
                            )

        timeline = getattr(design, "timeline", None)
        if timeline is not None and hasattr(timeline, "count"):
            for ti in range(timeline.count):
                t = timeline.item(ti)
                if t is None:
                    continue
                _collect(t, "feature", getattr(t, "name", None), _owner_token(t))
                feat = getattr(t, "entity", None)
                if feat is not None:
                    _collect(
                        feat, "feature", getattr(feat, "name", None), _owner_token(feat)
                    )

        return make_result(
            data={
                "group": group_filter,
                "records": records,
                "count": len(records),
                "candidates": candidates,
            },
            summary=f"Executed mutate:{op}",
            document=_doc_state(safe_doc, doc),
        )

    # Fallback for other mutate-group operations (geometry/style mutations, to be
    # owned by later tasks): the same authoritative guard + geometry primitive +
    # one-command metadata plan structure. When the Bridge attaches a metadata
    # plan, THIS same script execution applies it right after the geometry
    # primitive — provenance travels with the geometry change in one command,
    # with identical preflight/compensation/post-fingerprint atomicity. Without
    # a plan nothing metadata-related is touched.
    expected_fp = PAYLOAD.get("expected_fingerprint")
    current_fp, _canonical, doc_ref = collect_model_fingerprint(PAYLOAD)
    if expected_fp is None or expected_fp != current_fp:
        return make_error(
            "REVISION_CONFLICT",
            "Authoritative Fusion-side guard: model fingerprint diverged from baseline",
            details={
                "document_ref": doc_ref,
                "expected_revision": PAYLOAD.get("expected_revision"),
                "expected_fingerprint": expected_fp,
                "current_fingerprint": current_fp,
                "applied": False,
            },
        )

    _app, doc, design, safe_doc = _open_runtime()

    plan_writes = PAYLOAD.get("metadata_writes")
    plan_removals = PAYLOAD.get("metadata_removals")
    has_plan = isinstance(plan_writes, list) and isinstance(plan_removals, list)
    plan_owner = None
    plan_undo = None

    if has_plan:
        plan_owner, target_ref = _resolve_owner(
            doc, design, PAYLOAD.get("target")
        )
        _preflight_metadata_plan(plan_owner, plan_writes, plan_removals)

    primitive_hook = globals().get("_mutation_primitive")
    geometry_capture = None
    geometry_applied = False
    if has_plan:
        geometry_capture = _prepare_geometry_compensation(primitive_hook)
    try:
        if callable(primitive_hook):
            geometry_applied = True
            primitive_hook(PAYLOAD)

        if has_plan:
            applied_writes, applied_removals, plan_undo = _apply_metadata_plan(
                plan_owner, plan_writes, plan_removals
            )
            if not applied_writes:
                _rollback_metadata(plan_owner, plan_undo)
                raise FusionScriptError(
                    "INVALID_ARGUMENT",
                    "Metadata plan produced no attribute writes; provenance cannot be recorded transactionally",
                    details={"operation": op},
                )

        post_fp, _post_canonical, post_doc_ref = collect_model_fingerprint(PAYLOAD)
    except Exception as exc:
        try:
            if plan_undo is not None:
                _rollback_metadata(plan_owner, plan_undo)
        finally:
            if geometry_applied and has_plan:
                _compensate_geometry(geometry_capture, current_fp)
        if isinstance(exc, FusionScriptError):
            raise exc
        raise FusionScriptError(
            "FUSION_API_ERROR",
            "Mutation execution failed after an apply attempt; exact compensation completed",
            details={"operation": op, "applied": False, "compensated": True},
        )
    if not post_fp or not isinstance(post_fp, str) or not post_fp.strip():
        try:
            _rollback_metadata(plan_owner, plan_undo)
        finally:
            if geometry_applied and has_plan:
                _compensate_geometry(geometry_capture, current_fp)
        return make_error(
            "FINGERPRINT_FAILED",
            "Authoritative post-apply model fingerprint collection failed or returned empty",
            details={
                "document_ref": post_doc_ref or doc_ref,
                "applied": False,
                "compensated": True,
            },
        )
    if post_fp == current_fp:
        try:
            _rollback_metadata(plan_owner, plan_undo)
        finally:
            if geometry_applied and has_plan:
                _compensate_geometry(geometry_capture, current_fp)
        return make_error(
            "MUTATION_FAILED",
            "Mutation failed to produce distinct post-apply model state; cannot advance model revision",
            details={
                "document_ref": post_doc_ref or doc_ref,
                "initial_fingerprint": current_fp,
                "post_fingerprint": post_fp,
                "applied": False,
                "compensated": True,
            },
        )

    try:
        result_data = {
            "operation": op,
            "applied": True,
            "fingerprint": post_fp,
            "document_ref": post_doc_ref or doc_ref,
        }
        if has_plan:
            result_data["attributes"] = applied_writes
            result_data["removed"] = applied_removals
            result_data["provenance"] = PAYLOAD.get("provenance")
        return make_result(
            data=result_data,
            summary=f"Executed mutate:{op}",
            document=_doc_state(safe_doc, doc),
            changed_refs=[target_ref] if has_plan and target_ref else [],
        )
    except Exception as exc:
        try:
            if plan_undo is not None:
                _rollback_metadata(plan_owner, plan_undo)
        finally:
            if geometry_applied and has_plan:
                _compensate_geometry(geometry_capture, current_fp)
        if isinstance(exc, FusionScriptError):
            raise exc
        raise FusionScriptError(
            "FUSION_API_ERROR",
            "Mutation execution failed after an apply attempt; exact compensation completed",
            details={"operation": op, "applied": False, "compensated": True},
        )
