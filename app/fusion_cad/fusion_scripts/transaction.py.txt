def run():
    op = PAYLOAD.get("operation", "")

    if op == "begin":
        expected_fp = PAYLOAD.get("expected_fingerprint")
        current_fp, canonical_data, doc_ref = collect_model_fingerprint(PAYLOAD)

        if not current_fp or not isinstance(current_fp, str) or not current_fp.strip():
            return make_error(
                "FINGERPRINT_FAILED",
                "Authoritative model fingerprint collection failed or returned empty at transaction begin",
                details={"document_ref": doc_ref, "applied": False},
            )

        if not doc_ref or not isinstance(doc_ref, str) or not doc_ref.strip():
            return make_error(
                "NO_ACTIVE_DESIGN",
                "Document lacks stable runtime identity at transaction begin",
                details={"applied": False},
            )

        if expected_fp is not None and expected_fp != current_fp:
            return make_error(
                "REVISION_CONFLICT",
                f"Authoritative Fusion-side guard: model fingerprint diverged from baseline during transaction {op}",
                details={
                    "document_ref": doc_ref,
                    "expected_revision": PAYLOAD.get("expected_revision"),
                    "expected_fingerprint": expected_fp,
                    "current_fingerprint": current_fp,
                    "applied": False,
                },
            )

        return make_result(
            data={
                "operation": op,
                "applied": True,
                "fingerprint": current_fp,
                "document_ref": doc_ref,
                "transaction_id": PAYLOAD.get("transaction_id"),
            },
            summary=f"Executed transaction:{op}",
        )

    if op in ("preview", "commit"):
        expected_fp = PAYLOAD.get("expected_fingerprint")
        current_fp, canonical_data, doc_ref = collect_model_fingerprint(PAYLOAD)

        if not current_fp or not isinstance(current_fp, str) or not current_fp.strip():
            return make_error(
                "FINGERPRINT_FAILED",
                f"Authoritative model fingerprint collection failed or returned empty during transaction {op}",
                details={"document_ref": doc_ref, "applied": False},
            )

        if not doc_ref or not isinstance(doc_ref, str) or not doc_ref.strip():
            return make_error(
                "NO_ACTIVE_DESIGN",
                f"Document lacks stable runtime identity during transaction {op}",
                details={"applied": False},
            )

        if expected_fp is None or expected_fp != current_fp:
            return make_error(
                "REVISION_CONFLICT",
                f"Authoritative Fusion-side guard: model fingerprint diverged from baseline during transaction {op}",
                details={
                    "document_ref": doc_ref,
                    "expected_revision": PAYLOAD.get("expected_revision"),
                    "expected_fingerprint": expected_fp,
                    "current_fingerprint": current_fp,
                    "applied": False,
                },
            )

        if op == "commit":
            commit_primitive = globals().get("_transaction_commit_primitive")
            if callable(commit_primitive):
                commit_primitive(PAYLOAD)

            post_fp, _, post_doc_ref = collect_model_fingerprint(PAYLOAD)
            if not post_fp or not isinstance(post_fp, str) or not post_fp.strip():
                return make_error(
                    "FINGERPRINT_FAILED",
                    "Authoritative post-apply fingerprint collection failed or returned empty after transaction commit",
                    details={"document_ref": post_doc_ref or doc_ref, "applied": False},
                )

            if not post_doc_ref or not isinstance(post_doc_ref, str) or not post_doc_ref.strip():
                return make_error(
                    "NO_ACTIVE_DESIGN",
                    "Document lacks stable runtime identity after transaction commit",
                    details={"applied": False},
                )

            expected_doc = PAYLOAD.get("document_ref")
            if expected_doc and expected_doc != post_doc_ref:
                return make_error(
                    "WRONG_DOCUMENT",
                    f"Active document {post_doc_ref} does not match expected document {expected_doc} after transaction commit",
                    details={
                        "document_ref": post_doc_ref,
                        "expected_document_ref": expected_doc,
                        "applied": False,
                    },
                )

            if post_fp == current_fp:
                return make_error(
                    "MUTATION_FAILED",
                    "Transaction commit failed to produce distinct post-apply model state; cannot advance model revision",
                    details={
                        "document_ref": post_doc_ref,
                        "initial_fingerprint": current_fp,
                        "post_fingerprint": post_fp,
                        "applied": False,
                    },
                )

            return make_result(
                data={
                    "operation": op,
                    "applied": True,
                    "fingerprint": post_fp,
                    "document_ref": post_doc_ref,
                    "transaction_id": PAYLOAD.get("transaction_id"),
                },
                summary=f"Executed transaction:{op}",
            )
        elif op == "preview":
            preview_primitive = globals().get("_transaction_preview_primitive")
            if callable(preview_primitive):
                preview_primitive(PAYLOAD)

            return make_result(
                data={
                    "operation": op,
                    "applied": False,
                    "fingerprint": current_fp,
                    "document_ref": doc_ref,
                    "transaction_id": PAYLOAD.get("transaction_id"),
                    "preview": True,
                },
                summary=f"Executed transaction:{op}",
                diff={"status": "clean", "changes": []},
                validation={"status": "passed", "errors": []},
            )

    if op in ("abort", "rollback"):
        abort_primitive = globals().get(f"_transaction_{op}_primitive")
        if callable(abort_primitive):
            abort_primitive(PAYLOAD)

        post_fp, canonical_data, post_doc_ref = collect_model_fingerprint(PAYLOAD)
        if not post_fp or not isinstance(post_fp, str) or not post_fp.strip():
            return make_error(
                "FINGERPRINT_FAILED",
                f"Authoritative model fingerprint collection failed or returned empty during transaction {op}",
                details={"document_ref": post_doc_ref, "applied": False},
            )

        if not post_doc_ref or not isinstance(post_doc_ref, str) or not post_doc_ref.strip():
            return make_error(
                "NO_ACTIVE_DESIGN",
                f"Document lacks stable runtime identity during transaction {op}",
                details={"applied": False},
            )

        expected_doc = PAYLOAD.get("document_ref")
        if expected_doc and expected_doc != post_doc_ref:
            return make_error(
                "WRONG_DOCUMENT",
                f"Active document {post_doc_ref} does not match expected document {expected_doc} during transaction {op}",
                details={
                    "document_ref": post_doc_ref,
                    "expected_document_ref": expected_doc,
                    "applied": False,
                },
            )

        return make_result(
            data={
                "operation": op,
                "applied": True,
                "fingerprint": post_fp,
                "document_ref": post_doc_ref,
                "transaction_id": PAYLOAD.get("transaction_id"),
            },
            summary=f"Executed transaction:{op}",
        )

    return make_result(data={"operation": op}, summary=f"Executed transaction:{op}")
