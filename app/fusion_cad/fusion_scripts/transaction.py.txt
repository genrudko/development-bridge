def _transaction_fingerprint(payload):
    hook = globals().get("_transaction_fingerprint_primitive")
    return hook(payload) if callable(hook) else collect_model_fingerprint(payload)


def _minimal_preview_diff(before, after):
    bc, ac = dict(before.get("counts") or {}), dict(after.get("counts") or {})
    br, ar = set(before.get("refs") or []), set(after.get("refs") or [])
    return {
        "before": {"structural_hash": before.get("structural_hash"), "counts": bc},
        "after": {"structural_hash": after.get("structural_hash"), "counts": ac},
        "counts": {key: int(ac.get(key, 0)) - int(bc.get(key, 0)) for key in sorted(set(bc) | set(ac))},
        "refs": {"added": sorted(ar - br), "removed": sorted(br - ar)},
    }


def _require_transaction_primitive(name):
    primitive = globals().get(name)
    if not callable(primitive):
        raise FusionScriptError("CAPABILITY_UNAVAILABLE", "Verified transaction runtime primitive is unavailable", {"primitive": name, "applied": False})
    return primitive


def _transaction_opaque_refs(value):
    import re
    refs = value if isinstance(value, list) else []
    if any(not isinstance(ref, str) or re.fullmatch(r"ent_[A-Za-z0-9._-]+", ref) is None for ref in refs):
        raise FusionScriptError("FUSION_API_ERROR", "Transaction runtime returned a non-opaque generated reference", {"applied": False})
    return list(refs)


def _transaction_exact_text_create_plan(plan):
    if not isinstance(plan, list) or len(plan) != 1 or not isinstance(plan[0], dict) or plan[0].get("action_type") != "text_create":
        raise FusionScriptError("CAPABILITY_UNAVAILABLE", "P0 transaction replay supports exactly one staged text_create action", {"applied": False})
    action = plan[0]
    position = action.get("position")
    frame = position.get("frame") if isinstance(position, dict) else None
    if not isinstance(position, dict) or not isinstance(frame, dict) or frame.get("space") != "world" or abs(float(position.get("z", 0.0))) > 1e-6:
        raise FusionScriptError("CAPABILITY_UNAVAILABLE", "P0 transaction text_create supports only the verified world XY plane (z=0)", {"applied": False})
    if action.get("font", "Arial") != "Arial" or action.get("alignment", "left") != "left" or action.get("flip_x", False) is not False or action.get("flip_y", False) is not False or action.get("target_plane_or_face") is not None:
        raise FusionScriptError("CAPABILITY_UNAVAILABLE", "P0 transaction text_create supports only default font, alignment, flips, and target plane options", {"applied": False})
    writes = action.get("metadata_writes")
    provenance = action.get("provenance")
    if not isinstance(writes, list) or len(writes) != 1 or not isinstance(writes[0], dict) or writes[0].get("name") != "provenance" or not isinstance(writes[0].get("value"), str) or not isinstance(provenance, dict):
        raise FusionScriptError("CAPABILITY_UNAVAILABLE", "P0 transaction text_create requires one service-prepared provenance write", {"applied": False})
    try:
        persisted_record = json.loads(writes[0]["value"])
    except Exception as exc:
        raise FusionScriptError("INVALID_ARGUMENT", "Service-prepared provenance is not valid JSON", {"applied": False}) from exc
    if persisted_record != provenance:
        raise FusionScriptError("INVALID_ARGUMENT", "Service-prepared provenance write and record disagree", {"applied": False})
    return action


def _transaction_semantic_hash(canonical):
    """Token-insensitive deterministic hash for preview/commit equivalence only."""
    import hashlib

    volatile = {"id", "native_token", "owner_id", "document_ref"}
    unordered_top = {"timeline", "components", "occurrences", "bodies", "sketches", "parameters", "attributes", "sketch_texts"}

    def scrub(value):
        if isinstance(value, dict):
            return {key: scrub(val) for key, val in sorted(value.items()) if key not in volatile}
        if isinstance(value, tuple):
            return [scrub(item) for item in value]
        if isinstance(value, list):
            return [scrub(item) for item in value]
        return canonicalize_value(value)

    source = canonical if isinstance(canonical, dict) else {}
    semantic = {}
    for key, value in sorted(source.items()):
        clean = scrub(value)
        if key in unordered_top and isinstance(clean, list):
            clean = sorted(clean, key=lambda item: json.dumps(item, sort_keys=True, ensure_ascii=False, separators=(",", ":")))
        semantic[key] = clean
    raw = json.dumps(semantic, sort_keys=True, ensure_ascii=False, separators=(",", ":"))
    return "sha256:" + hashlib.sha256(raw.encode("utf-8")).hexdigest()


def _transaction_snapshot_from_fingerprint(fingerprint, canonical):
    canonical = canonical if isinstance(canonical, dict) else {}
    counts = {}
    for key in ("timeline", "components", "occurrences", "bodies", "sketches", "parameters", "attributes"):
        value = canonical.get(key)
        if isinstance(value, (list, tuple, dict)):
            counts[key] = len(value)
    return {"structural_hash": _transaction_semantic_hash(canonical), "counts": counts, "refs": []}


def _transaction_replay_signature(snapshot, applied):
    diff = _minimal_preview_diff(PAYLOAD.get("baseline_snapshot") or {}, snapshot)
    return {
        "plan_hash": PAYLOAD.get("plan_hash"),
        "structural_hash": snapshot.get("structural_hash"),
        "mutation_counts": diff["counts"],
        "provenance": {
            "identity": applied.get("provenance"),
            "persisted": applied.get("persisted_provenance", applied.get("provenance")),
            "same_operation": applied.get("same_operation_provenance") is True,
        },
    }


def _transaction_apply_text_create_mutation(root, action, preview):
    """Apply the exact P0 text_create mutation with same-operation provenance.

    Shared by the primary PTransaction runtime and the explicit legacy
    CommandDefinition fallback so both paths reuse the IDENTICAL mutation and
    provenance semantics: world-XY SketchText creation (the exact-plan and
    world-XY constraints are enforced by _transaction_exact_text_create_plan),
    the service-prepared provenance write/readback, fingerprint snapshot +
    validation, and the adapter-private native ref hint ONLY on commit (preview
    evidence stays non-durable and never carries native tokens outward).
    """
    import adsk.core

    position = action["position"]
    sketch = root.sketches.add(root.xYConstructionPlane)
    point = adsk.core.Point3D.create(float(position["x"]) / 10.0, float(position["y"]) / 10.0, 0.0)
    text_input = sketch.sketchTexts.createInput(action["text"], float(action["height_mm"]) / 10.0, point)
    text = sketch.sketchTexts.add(text_input)
    write = action["metadata_writes"][0]
    text.attributes.add("bridge.cad/v1", "provenance", write["value"])
    actual = text.attributes.itemByName("bridge.cad/v1", "provenance")
    actual_value = getattr(actual, "value", None) if actual else None
    if actual_value != write["value"] or json.loads(actual_value) != action["provenance"]:
        raise FusionScriptError("FUSION_API_ERROR", "Persisted SketchText provenance read-back did not match the service-prepared value", {"applied": False})
    during_fp, during_canonical, _during_doc = _transaction_fingerprint(PAYLOAD)
    collector = globals().get("_transaction_validation_evidence_primitive")
    validation = collector(PAYLOAD) if callable(collector) else collect_p0_validation_evidence(PAYLOAD, during_canonical, _during_doc)
    if not isinstance(validation, dict):
        raise FusionScriptError("CAPABILITY_UNAVAILABLE", "Validation evidence is unavailable", {"applied": False})
    applied = {
        "provenance": action["provenance"],
        "persisted_provenance": json.loads(actual_value),
        "same_operation_provenance": True,
    }
    if not preview:
        applied["internal_ref_hints"] = [{"kind": "sketch_text", "native_token": str(text.entityToken)}]
    return {
        "snapshot": _transaction_snapshot_from_fingerprint(during_fp, during_canonical),
        "validation": validation,
        "applied": applied,
    }


def _transaction_native_text_command(plan, preview):
    import time
    import uuid

    import adsk
    import adsk.core
    import adsk.fusion

    action = _transaction_exact_text_create_plan(plan)
    app = adsk.core.Application.get()
    ui = getattr(app, "userInterface", None) if app else None
    design = adsk.fusion.Design.cast(getattr(app, "activeProduct", None)) if app else None
    root = getattr(design, "rootComponent", None) if design else None
    definitions = getattr(ui, "commandDefinitions", None) if ui else None
    if root is None or definitions is None:
        raise FusionScriptError("CAPABILITY_UNAVAILABLE", "Verified Fusion command transaction runtime is unavailable", {"applied": False})

    result = {
        "error": None,
        "applied": None,
        "snapshot": None,
        "validation": None,
        "command": None,
        "destroyed": False,
    }
    retained_handlers = []

    class ExecuteHandler(adsk.core.CommandEventHandler):
        def __init__(self):
            super().__init__()

        def notify(self, args):
            try:
                result.update(_transaction_apply_text_create_mutation(root, action, preview))
                if preview:
                    args.executeFailed = True
            except Exception as exc:
                result["error"] = exc
                args.executeFailed = True

    class DestroyHandler(adsk.core.CommandEventHandler):
        def __init__(self):
            super().__init__()

        def notify(self, _args):
            result["destroyed"] = True

    class CreatedHandler(adsk.core.CommandCreatedEventHandler):
        def __init__(self):
            super().__init__()

        def notify(self, args):
            command = args.command
            command.isAutoExecute = False
            execute_handler = ExecuteHandler()
            destroy_handler = DestroyHandler()
            retained_handlers.extend([execute_handler, destroy_handler])
            command.execute.add(execute_handler)
            command.destroy.add(destroy_handler)
            result["command"] = command

    command_id = "bridge_tx_text_" + uuid.uuid4().hex
    definition = definitions.addButtonDefinition(command_id, "Bridge transaction text", "Atomic P0 transaction text replay")
    if definition is None:
        raise FusionScriptError("FUSION_API_ERROR", "Fusion command definition creation failed", {"applied": False})
    delete_ok = False
    try:
        created_handler = CreatedHandler()
        retained_handlers.append(created_handler)
        definition.commandCreated.add(created_handler)
        if definition.execute() is not True:
            raise FusionScriptError("FUSION_API_ERROR", "Fusion command definition execute failed", {"applied": False})

        # CommandDefinition.execute queues commandCreated. Never call Command.doExecute
        # re-entrantly from commandCreated; wait until creation has fully returned.
        for _ in range(200):
            adsk.doEvents()
            if result["command"] is not None:
                break
            time.sleep(0.005)
        command = result["command"]
        if command is None:
            raise FusionScriptError("FUSION_API_ERROR", "Fusion command creation did not complete", {"applied": False})

        if command.doExecute(True) is not True:
            raise FusionScriptError("FUSION_API_ERROR", "Fusion command execution failed", {"applied": False})

        # Let Fusion finish commit/abort bookkeeping and command destruction before
        # inspecting/restoring state or deleting the API-created definition.
        for _ in range(200):
            adsk.doEvents()
            if result["destroyed"]:
                break
            time.sleep(0.005)
        if not result["destroyed"]:
            raise FusionScriptError("FUSION_API_ERROR", "Fusion command did not terminate cleanly", {"applied": False})

        # Live Fusion finalizes executeFailed rollback one main-loop turn after
        # Command.destroy. Drain a few bounded turns before deleting the
        # definition or letting the caller fingerprint authoritative state.
        for _ in range(3):
            adsk.doEvents()
            time.sleep(0.005)
    finally:
        delete_ok = bool(definition.deleteMe())
    if not delete_ok:
        raise FusionScriptError("FUSION_API_ERROR", "Fusion command definition cleanup failed", {"applied": False})
    if result["error"] is not None:
        error = result["error"]
        if isinstance(error, FusionScriptError):
            raise error
        raise FusionScriptError("FUSION_API_ERROR", "Fusion command transaction text replay failed", {"applied": False}) from error
    if not isinstance(result["applied"], dict):
        raise FusionScriptError("FUSION_API_ERROR", "Fusion command transaction did not execute", {"applied": False})
    return result


# Fixed safe transaction boundary name for the primary PTransaction runtime.
# Literal only: user/untrusted text is NEVER interpolated here.
_PTRANSACTION_STATIC_NAME = "bridge_cad_transaction"


def _transaction_ptx_command(plan, preview):
    """Apply the exact staged plan inside a proven PTransaction boundary.

    Every native boundary command must return the live-proven success marker
    ``"1"``. A rejected Start fails before mutation. Once Start has been
    proven successful, any terminal action whose outcome cannot also be proven
    is OPERATION_UNCERTAIN and is never retried. Mutation failures get exactly
    one cleanup Abort; the original error is re-raised only when that Abort is
    proven successful.
    """
    import adsk
    import adsk.core
    import adsk.fusion

    action = _transaction_exact_text_create_plan(plan)
    app = adsk.core.Application.get()
    design = adsk.fusion.Design.cast(getattr(app, "activeProduct", None)) if app else None
    root = getattr(design, "rootComponent", None) if design else None
    execute_text_command = getattr(app, "executeTextCommand", None) if app is not None else None
    if root is None or not callable(execute_text_command):
        raise FusionScriptError("CAPABILITY_UNAVAILABLE", "Verified PTransaction runtime is unavailable", {"applied": False})

    try:
        started = execute_text_command('PTransaction.Start "' + _PTRANSACTION_STATIC_NAME + '"')
    except BaseException:
        raise FusionScriptError(
            "OPERATION_UNCERTAIN",
            "PTransaction start outcome is uncertain",
        ) from None
    if str(started).strip() != "1":
        raise FusionScriptError(
            "CAPABILITY_UNAVAILABLE",
            "PTransaction start was not accepted",
            {"applied": False},
        )

    def abort_once():
        try:
            return str(execute_text_command("PTransaction.Abort")).strip() == "1"
        except BaseException:
            return False

    try:
        result = _transaction_apply_text_create_mutation(root, action, preview)
    except BaseException:
        if not abort_once():
            raise FusionScriptError(
                "OPERATION_UNCERTAIN",
                "PTransaction cleanup abort outcome is uncertain",
            ) from None
        raise

    if preview:
        if not abort_once():
            raise FusionScriptError(
                "OPERATION_UNCERTAIN",
                "PTransaction preview abort outcome is uncertain",
            )
    else:
        try:
            committed = execute_text_command("PTransaction.Commit")
        except BaseException:
            raise FusionScriptError(
                "OPERATION_UNCERTAIN",
                "PTransaction commit outcome is uncertain",
            ) from None
        if str(committed).strip() != "1":
            raise FusionScriptError(
                "OPERATION_UNCERTAIN",
                "PTransaction commit outcome is uncertain",
            )
    return result


def _transaction_runtime_selector():
    selector = PAYLOAD.get("_transaction_runtime")
    if selector not in (None, "ptransaction", "command_definition"):
        raise FusionScriptError(
            "INVALID_ARGUMENT",
            "Unknown internal transaction runtime selector",
            {"applied": False},
        )
    return selector


def _transaction_native_runtime(plan, preview):
    """Select the primary PTransaction runtime or explicit legacy fallback.

    After bounded live acceptance, an absent selector is intentionally equivalent
    to ``ptransaction``. The previous CommandDefinition/doExecute runtime remains
    available only through the internal ``command_definition`` selector while the
    final service-level acceptance is pending. Unknown selectors fail closed.
    """
    selector = _transaction_runtime_selector()
    if selector in (None, "ptransaction"):
        return _transaction_ptx_command(plan, preview)
    return _transaction_native_text_command(plan, preview)


def _transaction_guard(op):
    current_fp, _canonical, doc_ref = _transaction_fingerprint(PAYLOAD)
    if not isinstance(current_fp, str) or not current_fp.strip():
        raise FusionScriptError("FINGERPRINT_FAILED", "Authoritative fingerprint unavailable", {"applied": False})
    if not isinstance(doc_ref, str) or not doc_ref.strip():
        raise FusionScriptError("NO_ACTIVE_DESIGN", "Stable document identity unavailable", {"applied": False})
    expected_fp = PAYLOAD.get("expected_fingerprint")
    if op in ("preview", "commit") and expected_fp != current_fp:
        return None, make_error("REVISION_CONFLICT", "Authoritative Fusion-side transaction guard rejected a stale baseline", {"document_ref": doc_ref, "expected_revision": PAYLOAD.get("expected_revision"), "expected_fingerprint": expected_fp, "current_fingerprint": current_fp, "applied": False})
    return (current_fp, doc_ref), None


def _transaction_post_commit_state(current_fp, doc_ref):
    try:
        post_fp, _canonical, post_doc = _transaction_fingerprint(PAYLOAD)
    except BaseException:
        raise FusionScriptError(
            "OPERATION_UNCERTAIN",
            "Transaction commit verification outcome is uncertain",
        ) from None
    if post_doc != doc_ref or not isinstance(post_fp, str) or not post_fp.strip() or post_fp == current_fp:
        raise FusionScriptError(
            "OPERATION_UNCERTAIN",
            "Transaction commit verification outcome is uncertain",
        )
    return post_fp, post_doc


def run(_context=None):
    op = PAYLOAD.get("operation", "")
    if op not in ("begin", "stage", "status", "preview", "commit", "abort", "rollback"):
        return make_result(data={"operation": op}, summary="Executed transaction:" + str(op))
    _transaction_runtime_selector()
    guarded, error = _transaction_guard(op)
    if error:
        return error
    current_fp, doc_ref = guarded
    tx_id = PAYLOAD.get("transaction_id")

    if op == "begin":
        expected_fp = PAYLOAD.get("expected_fingerprint")
        if expected_fp is not None and expected_fp != current_fp:
            return make_error("REVISION_CONFLICT", "Authoritative Fusion-side transaction guard rejected a stale baseline", {"document_ref": doc_ref, "current_fingerprint": current_fp, "applied": False})
        return make_result(data={"operation": op, "applied": True, "fingerprint": current_fp, "document_ref": doc_ref, "transaction_id": tx_id}, summary="Executed transaction:begin")
    if op in ("stage", "status"):
        return make_result(data={"operation": op, "applied": True, "fingerprint": current_fp, "document_ref": doc_ref, "transaction_id": tx_id}, summary="Executed transaction:" + op)

    plan = PAYLOAD.get("plan")
    if op in ("preview", "commit") and (not isinstance(plan, list) or not plan):
        raise FusionScriptError("INVALID_ARGUMENT", "Transaction requires a non-empty declarative plan", {"applied": False})

    if op == "preview":
        if not callable(globals().get("_transaction_begin_primitive")):
            native = _transaction_native_runtime(plan, True)
            restored_fp, _, restored_doc = _transaction_fingerprint(PAYLOAD)
            if restored_fp != current_fp or restored_doc != doc_ref:
                raise FusionScriptError("TRANSACTION_CONFLICT", "Preview command abort did not restore the authoritative baseline", {"applied": False})
            signature = _transaction_replay_signature(native["snapshot"], native["applied"])
            return make_result(data={"operation": op, "applied": False, "preview": True, "fingerprint": current_fp, "document_ref": doc_ref, "transaction_id": tx_id, "preview_snapshot": native["snapshot"], "replay_signature": signature, "preview_refs": [], "preview_refs_durable": False, "provenance": native["applied"].get("provenance"), "provenance_durable": False}, summary="Previewed and aborted exact staged transaction plan", diff=_minimal_preview_diff(PAYLOAD.get("baseline_snapshot") or {}, native["snapshot"]), validation=native["validation"])
        begin = _require_transaction_primitive("_transaction_begin_primitive")
        apply_plan = _require_transaction_primitive("_transaction_apply_plan_primitive")
        snapshot = _require_transaction_primitive("_transaction_snapshot_primitive")
        validate = _require_transaction_primitive("_transaction_validate_primitive")
        abort = _require_transaction_primitive("_transaction_abort_primitive")
        begin(PAYLOAD)
        try:
            applied = apply_plan(plan)
            preview_snapshot = snapshot(PAYLOAD)
            validation = validate(PAYLOAD)
        finally:
            abort(PAYLOAD)
        restored_fp, _, restored_doc = _transaction_fingerprint(PAYLOAD)
        if restored_fp != current_fp or restored_doc != doc_ref:
            raise FusionScriptError("TRANSACTION_CONFLICT", "Preview abort did not restore the authoritative baseline", {"applied": False})
        preview_refs = _transaction_opaque_refs(applied.get("refs")) if isinstance(applied, dict) else []
        return make_result(data={"operation": op, "applied": False, "preview": True, "fingerprint": current_fp, "document_ref": doc_ref, "transaction_id": tx_id, "preview_snapshot": preview_snapshot, "replay_signature": _transaction_replay_signature(preview_snapshot, applied), "preview_refs": preview_refs, "preview_refs_durable": False, "provenance": applied.get("provenance") if isinstance(applied, dict) else None, "provenance_durable": False}, summary="Previewed and aborted exact staged transaction plan", diff=_minimal_preview_diff(PAYLOAD.get("baseline_snapshot") or {}, preview_snapshot), validation=validation)

    if op == "commit":
        if not callable(globals().get("_transaction_begin_primitive")):
            native = _transaction_native_runtime(plan, False)
            applied = native["applied"]
            post_fp, post_doc = _transaction_post_commit_state(current_fp, doc_ref)
            signature = _transaction_replay_signature(native["snapshot"], applied)
            return make_result(data={"operation": op, "applied": True, "fingerprint": post_fp, "document_ref": post_doc, "transaction_id": tx_id, "refs": [], "internal_ref_hints": applied["internal_ref_hints"], "provenance": applied["provenance"], "persisted_provenance": applied["persisted_provenance"], "same_command_provenance": True, "same_operation_provenance": True, "replay_signature": signature, "replayed_plan": plan}, summary="Committed exact staged transaction plan", changed_refs=[])
        begin = _require_transaction_primitive("_transaction_begin_primitive")
        apply_plan = _require_transaction_primitive("_transaction_apply_plan_primitive")
        commit = _require_transaction_primitive("_transaction_commit_primitive")
        begin(PAYLOAD)
        applied = apply_plan(plan)
        if not isinstance(applied, dict) or applied.get("same_operation_provenance") is not True:
            raise FusionScriptError("FUSION_API_ERROR", "Transaction replay did not prove same-command provenance", {"applied": False})
        refs = _transaction_opaque_refs(applied.get("refs"))
        snapshot = _require_transaction_primitive("_transaction_snapshot_primitive")(PAYLOAD)
        commit(PAYLOAD)
        post_fp, post_doc = _transaction_post_commit_state(current_fp, doc_ref)
        return make_result(data={"operation": op, "applied": True, "fingerprint": post_fp, "document_ref": post_doc, "transaction_id": tx_id, "refs": refs, "provenance": applied.get("provenance"), "same_command_provenance": True, "replay_signature": _transaction_replay_signature(snapshot, applied), "replayed_plan": plan}, summary="Committed exact staged transaction plan", changed_refs=refs)

    if op in ("abort", "rollback"):
        primitive = globals().get("_transaction_" + op + "_primitive")
        if callable(primitive):
            primitive(PAYLOAD)
        post_fp, _, post_doc = _transaction_fingerprint(PAYLOAD)
        return make_result(data={"operation": op, "applied": True, "fingerprint": post_fp, "document_ref": post_doc, "transaction_id": tx_id}, summary="Executed transaction:" + op)
    raise FusionScriptError("INVALID_ARGUMENT", "Unsupported transaction operation", {"applied": False})
