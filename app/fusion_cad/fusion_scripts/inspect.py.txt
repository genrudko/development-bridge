import math


def _get_app_design():
    app = None
    doc = None
    design = None
    try:
        import adsk.core
        import adsk.fusion
        app = adsk.core.Application.get()
        if app:
            doc = getattr(app, "activeDocument", None)
            if doc and hasattr(doc, "products"):
                design = doc.products.itemByClass("adsk::fusion::Design")
    except Exception:
        pass
    if not doc or not design:
        raise FusionScriptError(
            "NO_ACTIVE_DESIGN",
            "Active document and design required for inspection",
            details={"has_doc": bool(doc), "has_design": bool(design)},
        )
    return app, doc, design


def _make_point(x, y, z):
    try:
        import adsk.core
        return adsk.core.Point3D.create(float(x), float(y), float(z))
    except Exception:
        class _P:
            def __init__(self, x, y, z):
                self.x = float(x)
                self.y = float(y)
                self.z = float(z)
        return _P(float(x), float(y), float(z))


def _coord_value(pt, name):
    v = getattr(pt, name, None)
    if v is None or isinstance(v, bool) or not isinstance(v, (int, float)):
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            f"Inspection requires exact finite '{name}' coordinate; missing or non-numeric geometry cannot be measured",
            details={"missing": name},
        )
    f = float(v)
    if not math.isfinite(f):
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            f"Inspection requires exact finite '{name}' coordinate; NaN/infinite geometry cannot be measured",
            details={"missing": name},
        )
    return f


def _point_dict(pt, frame=None):
    if pt is None:
        return None
    # Fusion native geometry is cm-based; public output is mm.
    return {
        "x": round(_mm(_coord_value(pt, "x")), 6),
        "y": round(_mm(_coord_value(pt, "y")), 6),
        "z": round(_mm(_coord_value(pt, "z")), 6),
        "frame": frame or {"space": "world", "ref": None},
    }


def _normalize_target(target):
    if isinstance(target, str):
        return {"ref": target}
    if isinstance(target, dict):
        return dict(target)
    return None


def _world_frame_or_fail(payload):
    """Inspection reports geometry in model/world space.

    A request asking for any other CoordinateFrame requires an exact verified
    transform that P0 inspection does not silently fabricate; fail closed instead.
    """
    f = payload.get("frame")
    if f is None:
        return
    if isinstance(f, dict) and f.get("space", "world") == "world":
        return
    raise FusionScriptError(
        "UNSUPPORTED_GEOMETRY",
        "Inspection reports geometry in model/world space; exact conversion to another coordinate frame requires a verified transform",
        details={"frame": f},
    )


def _vec3(pt):
    return [_coord_value(pt, "x"), _coord_value(pt, "y"), _coord_value(pt, "z")]


def _sub(a, b):
    return [a[0] - b[0], a[1] - b[1], a[2] - b[2]]


def _sub2(a, b):
    return [a[0] - b[0], a[1] - b[1]]


def _dot(a, b):
    return a[0] * b[0] + a[1] * b[1] + a[2] * b[2]


def _cross(a, b):
    return [
        a[1] * b[2] - a[2] * b[1],
        a[2] * b[0] - a[0] * b[2],
        a[0] * b[1] - a[1] * b[0],
    ]


def _norm(a):
    return math.sqrt(_dot(a, a))


def _normalize(a):
    n = _norm(a)
    if n == 0.0:
        return [0.0, 0.0, 0.0]
    return [a[0] / n, a[1] / n, a[2] / n]


def _mm(v):
    return float(v) * 10.0

def _mm2(v):
    return float(v) * 100.0

def _mm3(v):
    return float(v) * 1000.0

def _perpendicular(n):
    if abs(n[2]) < 0.9:
        u = _normalize(_cross([0.0, 0.0, 1.0], n))
    else:
        u = _normalize(_cross([1.0, 0.0, 0.0], n))
    return u


def _angle_deg(a, b):
    na = _norm(a)
    nb = _norm(b)
    if na == 0.0 or nb == 0.0:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Angle requires non-zero direction vectors",
        )
    cosv = max(-1.0, min(1.0, _dot(a, b) / (na * nb)))
    return math.degrees(math.acos(cosv))


def _entity_kind(entity):
    type_name = type(entity).__name__
    if type_name == "BRepBody" or (
        hasattr(entity, "isSolid") and hasattr(entity, "volume")
    ):
        return "body"
    if type_name == "BRepFace" or (
        hasattr(entity, "centroid")
        and hasattr(entity, "area")
        and hasattr(entity, "loops")
    ):
        return "face"
    if type_name == "BRepEdge" or (
        hasattr(entity, "length") and hasattr(entity, "startVertex")
    ):
        return "edge"
    if type_name == "Sketch" or hasattr(entity, "sketchCurves"):
        return "sketch"
    if type_name == "Occurrence" or hasattr(entity, "fullPathName"):
        return "occurrence"
    if type_name == "Component" or hasattr(entity, "bRepBodies"):
        return "component"
    return "other"


def _collect_candidates(design, kind, name):
    candidates = []
    root = getattr(design, "rootComponent", None)
    if root is None:
        return candidates
    if kind == "body":
        coll = getattr(root, "bRepBodies", None)
        if coll is not None:
            for i in range(coll.count):
                ent = coll.item(i)
                if name is None or getattr(ent, "name", None) == name:
                    candidates.append(ent)
    elif kind == "face":
        bodies = getattr(root, "bRepBodies", None)
        if bodies is not None:
            for i in range(bodies.count):
                body = bodies.item(i)
                faces = getattr(body, "faces", None)
                if faces is not None:
                    for j in range(faces.count):
                        f = faces.item(j)
                        if name is None or getattr(f, "name", None) == name:
                            candidates.append(f)
    elif kind == "edge":
        bodies = getattr(root, "bRepBodies", None)
        if bodies is not None:
            for i in range(bodies.count):
                body = bodies.item(i)
                edges = getattr(body, "edges", None)
                if edges is not None:
                    for j in range(edges.count):
                        e = edges.item(j)
                        if name is None or getattr(e, "name", None) == name:
                            candidates.append(e)
    elif kind == "sketch":
        coll = getattr(root, "sketches", None)
        if coll is not None:
            for i in range(coll.count):
                s = coll.item(i)
                if name is None or getattr(s, "name", None) == name:
                    candidates.append(s)
    elif kind == "sketch_curve":
        sketches = getattr(root, "sketches", None)
        if sketches is not None:
            for i in range(sketches.count):
                s = sketches.item(i)
                curves = getattr(s, "sketchCurves", None)
                if curves is not None:
                    for j in range(curves.count):
                        cv = curves.item(j)
                        if name is None or getattr(cv, "name", None) == name:
                            candidates.append(cv)
    elif kind == "occurrence":
        coll = getattr(root, "allOccurrences", None)
        if coll is not None:
            for i in range(coll.count):
                o = coll.item(i)
                if name is None or getattr(o, "name", None) == name:
                    candidates.append(o)
    elif kind == "component":
        comps = getattr(design, "allComponents", None)
        if comps is not None:
            for i in range(comps.count):
                c = comps.item(i)
                if name is None or getattr(c, "name", None) == name:
                    candidates.append(c)
    return candidates


def _resolve_entity(target):
    t = _normalize_target(target)
    if not t:
        raise FusionScriptError(
            "INVALID_ARGUMENT",
            "Inspection target must be a ref string or selector dict",
        )
    app, doc, design = _get_app_design()

    native_token = t.get("native_token")
    # A target carrying an authoritative native native_token hint (injected by the
    # service for a document-bounded opaque ref / resolved selector) must NEVER
    # fall back to kind/name substitution: a stale registered ref must fail closed
    # (REF_STALE) instead of silently resolving to an unrelated entity.
    has_authoritative_hint = "native_token" in t and t.get("ref") is not None

    if native_token:
        if not hasattr(design, "findEntityByToken"):
            if has_authoritative_hint:
                raise FusionScriptError(
                    "REF_STALE",
                    "Inspection target carried an authoritative native token but the active design cannot resolve native tokens; kind/name fallback is not permitted",
                    details={"ref": t.get("ref")},
                )
        else:
            resolved = None
            try:
                resolved = design.findEntityByToken(native_token)
            except FusionScriptError:
                raise
            except Exception:
                resolved = None
            if resolved is not None:
                if hasattr(resolved, "count"):
                    if resolved.count == 1:
                        return resolved.item(0)
                    if resolved.count > 1:
                        raise FusionScriptError(
                            "REF_SPLIT",
                            "Inspection target resolved to multiple split entities",
                            details={"ref": t.get("ref")},
                        )
                    # count == 0 -> zero native lookup; fail closed below.
                else:
                    return resolved
            # resolved is None or an empty native collection: native lookup failed
            # or returned zero candidates.

    if has_authoritative_hint:
        raise FusionScriptError(
            "REF_STALE",
            "Inspection target native token could not be resolved in the active document; kind/name fallback is not permitted for an authoritative native-hint target",
            details={"ref": t.get("ref")},
        )

    kind = t.get("kind")
    if not kind:
        raise FusionScriptError(
            "REF_STALE",
            "Inspection target could not be resolved without kind hints",
            details={"ref": t.get("ref")},
        )
    name = t.get("name")
    candidates = _collect_candidates(design, kind, name)
    if len(candidates) == 1:
        return candidates[0]
    if len(candidates) > 1:
        raise FusionScriptError(
            "REF_AMBIGUOUS",
            "Inspection target resolved to multiple ambiguous entities",
            details={"ref": t.get("ref"), "kind": kind},
        )
    raise FusionScriptError(
        "REF_STALE",
        "Inspection target could not be resolved in active document",
        details={"ref": t.get("ref"), "kind": kind},
    )


def _bbox_center(bb):
    min_pt = getattr(bb, "minPoint", None)
    max_pt = getattr(bb, "maxPoint", None)
    if min_pt is None or max_pt is None:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Bounding box center requires exact minPoint/maxPoint",
        )
    return _make_point(
        (min_pt.x + max_pt.x) / 2.0,
        (min_pt.y + max_pt.y) / 2.0,
        (min_pt.z + max_pt.z) / 2.0,
    )


def _bbox_dict(bb, frame=None):
    min_pt = getattr(bb, "minPoint", None)
    max_pt = getattr(bb, "maxPoint", None)
    if min_pt is None or max_pt is None:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Bounding box requires exact minPoint/maxPoint",
        )
    # Fusion native bounding boxes are cm-based; public output is mm.
    return {
        "min": [round(_mm(min_pt.x), 6), round(_mm(min_pt.y), 6), round(_mm(min_pt.z), 6)],
        "max": [round(_mm(max_pt.x), 6), round(_mm(max_pt.y), 6), round(_mm(max_pt.z), 6)],
        "frame": frame or {"space": "world", "ref": None},
    }


def _entity_centroid(entity):
    kind = _entity_kind(entity)
    if kind == "body":
        phys = getattr(entity, "physicalProperties", None)
        if phys is None:
            get_phys = getattr(entity, "getPhysicalProperties", None)
            if callable(get_phys):
                phys = get_phys()
        if phys is not None and getattr(phys, "centerOfMass", None) is not None:
            return phys.centerOfMass
        # No heuristic bbox-center fallback for an exact centroid.
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Body centroid requires exact physicalProperties centerOfMass",
        )
    if kind == "face":
        c = getattr(entity, "centroid", None)
        if c is not None:
            return c
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Face centroid requires exact centroid property",
        )
    if kind == "edge":
        # A centroid is exact only for a straight line edge (midpoint);
        # curved-edge centroids are never the endpoint midpoint.
        geom = getattr(entity, "geometry", None)
        curve_type = (
            getattr(geom, "curveType", None)
            or getattr(geom, "objectType", None)
        )
        if curve_type not in ("Line3D", "Line"):
            raise FusionScriptError(
                "UNSUPPORTED_GEOMETRY",
                "Edge centroid requires a straight line edge; curved edges unsupported",
            )
        sv = getattr(entity, "startVertex", None)
        ev = getattr(entity, "endVertex", None)
        if sv is not None and ev is not None:
            sg = getattr(sv, "geometry", None)
            eg = getattr(ev, "geometry", None)
            if sg is not None and eg is not None:
                return _make_point(
                    (sg.x + eg.x) / 2.0,
                    (sg.y + eg.y) / 2.0,
                    (sg.z + eg.z) / 2.0,
                )
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Edge centroid requires exact start/end vertex geometry",
        )
    if kind == "sketch":
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Sketch centroid has no exact Fusion centroid API; unsupported",
        )
    raise FusionScriptError(
        "UNSUPPORTED_GEOMETRY",
        "Centroid unsupported for this entity kind",
        details={"kind": kind},
    )


def _face_perimeter(face):
    total = 0.0
    loops = getattr(face, "loops", None)
    if loops is None or not hasattr(loops, "count"):
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Face perimeter requires readable loops collection",
        )
    for i in range(loops.count):
        loop = loops.item(i)
        edges = getattr(loop, "edges", None)
        if edges is None or not hasattr(edges, "count"):
            raise FusionScriptError(
                "UNSUPPORTED_GEOMETRY",
                "Face perimeter requires readable loop edges collection",
            )
        for j in range(edges.count):
            e = edges.item(j)
            length = getattr(e, "length", None)
            if length is None:
                raise FusionScriptError(
                    "UNSUPPORTED_GEOMETRY",
                    "Face perimeter requires exact edge length",
                )
            total += float(length)
    return float(_mm(total))


def _describe_entity(entity, ref, kind, name, native_type):
    frame = {"space": "world", "ref": None}
    measures = {}
    ekind = _entity_kind(entity)
    if ekind == "body":
        vol = getattr(entity, "volume", None)
        if vol is not None:
            measures["volume"] = {
                "quantity": "volume",
                "value": float(_mm3(vol)),
                "unit": "mm^3",
            }
        area = getattr(entity, "area", None)
        if area is not None:
            measures["area"] = {
                "quantity": "area",
                "value": float(_mm2(area)),
                "unit": "mm^2",
            }
        bb = getattr(entity, "boundingBox", None)
        if bb is not None:
            measures["bounding_box"] = _bbox_dict(bb, frame)
    elif ekind == "face":
        area = getattr(entity, "area", None)
        if area is not None:
            measures["area"] = {
                "quantity": "area",
                "value": float(_mm2(area)),
                "unit": "mm^2",
            }
        measures["perimeter"] = {
            "quantity": "perimeter",
            "value": _face_perimeter(entity),
            "unit": "mm",
        }
        bb = getattr(entity, "boundingBox", None)
        if bb is not None:
            measures["bounding_box"] = _bbox_dict(bb, frame)
    elif ekind == "edge":
        length = getattr(entity, "length", None)
        if length is not None:
            measures["length"] = {
                "quantity": "perimeter",
                "value": float(_mm(length)),
                "unit": "mm",
            }
    elif ekind == "sketch":
        bb = getattr(entity, "boundingBox", None)
        if bb is not None:
            measures["bounding_box"] = _bbox_dict(bb, frame)
    return {
        "target": {
            "ref": ref,
            "kind": kind,
            "name": name,
            "native_type": native_type,
        },
        "frame": frame,
        "measures": measures,
    }


def _curve_type_of(geom):
    """Best-effort exact curve-type string for a geometry object.

    Real Fusion Object3D exposes objectType (namespaced); some runtimes expose
    curveType; fake/other geometries fall back to the class name. The result is
    only ever used for a positive straight-line proof, never as a guess.
    """
    ct = getattr(geom, "curveType", None)
    if ct:
        return str(ct)
    ot = getattr(geom, "objectType", None)
    if ot:
        return str(ot)
    return type(geom).__name__


def _is_straight_line_curve(geom):
    """Positively prove a boundary curve is a straight line (Line3D / namespaced Line3D)."""
    ct = _curve_type_of(geom)
    if ct in ("Line3D", "Line"):
        return True
    return ct.endswith(".Line3D") or ct.endswith(".Line")


def _is_circular_curve(geom):
    """Positively prove a boundary curve is a circle or circular arc.

    Fusion legitimately exposes full circles as Circle3D and circular arcs as
    Arc3D (namespaced variants such as 'adsk.core.Circle3D' are accepted).
    Ellipse, elliptical-arc, spline, or unknown curve types are never proven
    circular here, so a center+normal alone is never enough.
    """
    ct = _curve_type_of(geom)
    if ct in ("Circle3D", "Arc3D", "Circle", "Arc"):
        return True
    return (
        ct.endswith(".Circle3D")
        or ct.endswith(".Arc3D")
        or ct.endswith(".Circle")
        or ct.endswith(".Arc")
    )


def _face_boundary_straight_edges(face):
    """Positively verify every boundary edge of the face is a straight line.

    A planar face whose entire boundary is straight line edges is a polygon
    (outer loop plus polygonal inner loops), so extrema of every linear
    function over the face occur at BRep vertices and vertex projection is
    exact. Curved edges (arcs/splines/ellipses) can have extrema between
    vertices; projecting only vertices would under-size/guess the box, so any
    curved or unverifiable boundary edge fails closed UNSUPPORTED_GEOMETRY.
    """
    loops = getattr(face, "loops", None)
    if loops is None or not hasattr(loops, "count"):
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Face oriented bounding box requires readable face loops to positively verify a straight-edged polygonal boundary",
        )
    if loops.count == 0:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Face oriented bounding box requires boundary loops; a face without loops cannot be verified as a straight-edged polygon",
        )
    for i in range(loops.count):
        loop = loops.item(i)
        edges = getattr(loop, "edges", None)
        if edges is None or not hasattr(edges, "count"):
            raise FusionScriptError(
                "UNSUPPORTED_GEOMETRY",
                "Face oriented bounding box requires readable loop edges to positively verify a straight-edged polygonal boundary",
            )
        for j in range(edges.count):
            edge = edges.item(j)
            edge_geom = getattr(edge, "geometry", None)
            if edge_geom is None:
                raise FusionScriptError(
                    "UNSUPPORTED_GEOMETRY",
                    "Face oriented bounding box requires exact edge geometry; an unverifiable boundary edge cannot be proven straight",
                )
            if not _is_straight_line_curve(edge_geom):
                raise FusionScriptError(
                    "UNSUPPORTED_GEOMETRY",
                    "Face oriented bounding box requires a provably straight-edged polygonal boundary; curved or unknown boundary edges can have extrema between vertices and cannot be vertex-projected exactly",
                    details={"curve_type": _curve_type_of(edge_geom)},
                )
    return True


def _face_oriented_bbox(face):
    geom = getattr(face, "geometry", None)
    if geom is None:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Face oriented bounding box requires surface geometry",
        )
    origin = getattr(geom, "origin", None)
    normal = getattr(geom, "normal", None)
    if origin is None or normal is None:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Face oriented bounding box requires planar origin and normal",
        )
    _face_boundary_straight_edges(face)
    n = _normalize(_vec3(normal))
    refv = getattr(geom, "referenceVector", None)
    if refv is not None:
        u = _normalize(_vec3(refv))
    else:
        u = _perpendicular(n)
    v = _cross(n, u)
    verts = getattr(face, "vertices", None)
    if verts is None or not hasattr(verts, "count") or verts.count == 0:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Face oriented bounding box requires readable vertices",
        )
    mins = [None, None, None]
    maxs = [None, None, None]
    for i in range(verts.count):
        vtx = verts.item(i)
        g = getattr(vtx, "geometry", None)
        if g is None:
            raise FusionScriptError(
                "UNSUPPORTED_GEOMETRY",
                "Face oriented bounding box requires exact vertex geometry",
            )
        rel = _sub(_vec3(g), _vec3(origin))
        coords = [_dot(rel, u), _dot(rel, v), _dot(rel, n)]
        for k in range(3):
            if mins[k] is None or coords[k] < mins[k]:
                mins[k] = coords[k]
            if maxs[k] is None or coords[k] > maxs[k]:
                maxs[k] = coords[k]
    center = [
        origin.x + u[0] * (mins[0] + maxs[0]) / 2.0 + v[0] * (mins[1] + maxs[1]) / 2.0 + n[0] * (mins[2] + maxs[2]) / 2.0,
        origin.y + u[1] * (mins[0] + maxs[0]) / 2.0 + v[1] * (mins[1] + maxs[1]) / 2.0 + n[1] * (mins[2] + maxs[2]) / 2.0,
        origin.z + u[2] * (mins[0] + maxs[0]) / 2.0 + v[2] * (mins[1] + maxs[1]) / 2.0 + n[2] * (mins[2] + maxs[2]) / 2.0,
    ]
    extents = [
        (maxs[0] - mins[0]) / 2.0,
        (maxs[1] - mins[1]) / 2.0,
        (maxs[2] - mins[2]) / 2.0,
    ]
    return {
        "center": _point_dict(_make_point(*center)),
        "axes": [[round(float(a), 6) for a in u], [round(float(a), 6) for a in v], [round(float(a), 6) for a in n]],
        "extents": [round(_mm(e), 6) for e in extents],
        "frame": {"space": "world", "ref": None},
    }


def _body_oriented_bbox(body):
    # The axis-aligned body.boundingBox is NOT an oriented bounding box: for a
    # rotated body the AABB is strictly larger than any exact oriented box and
    # relabeling it would be semantically wrong/guessed. P0 has no verified
    # exact Fusion oriented-bounding-box / measure API for an arbitrary body in
    # this static script architecture, so fail closed.
    raise FusionScriptError(
        "UNSUPPORTED_GEOMETRY",
        "Body oriented bounding box requires an exact Fusion oriented-bounding-box measure API; the axis-aligned body.boundingBox is not an oriented bounding box and is never returned as one",
        details={"kind": "body"},
    )

def _oriented_bbox(entity):
    ekind = _entity_kind(entity)
    if ekind == "face":
        return _face_oriented_bbox(entity)
    if ekind == "body":
        return _body_oriented_bbox(entity)
    raise FusionScriptError(
        "UNSUPPORTED_GEOMETRY",
        "Oriented bounding box unsupported for this entity kind",
        details={"kind": ekind},
    )


def _measure_distance(app, op, entity_a, entity_b):
    mm_mgr = getattr(app, "measureManager", None)
    if mm_mgr is None:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Distance requires an available measureManager; no centroid fallback",
        )
    try:
        if op == "minimum_distance" and hasattr(mm_mgr, "measureMinimumDistance"):
            res = mm_mgr.measureMinimumDistance(entity_a, entity_b)
        elif hasattr(mm_mgr, "measureDistance"):
            res = mm_mgr.measureDistance(entity_a, entity_b)
        else:
            res = None
    except Exception:
        res = None
    if res is None or getattr(res, "value", None) is None:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Distance requires an exact measureManager result",
        )
    from_pt = getattr(res, "pointOne", None) or getattr(res, "startPoint", None)
    to_pt = getattr(res, "pointTwo", None) or getattr(res, "endPoint", None)
    if from_pt is None or to_pt is None:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Distance requires exact witness points; no centroid fallback",
        )
    # measureManager value is native cm; public output is mm.
    return _mm(float(res.value)), from_pt, to_pt


def _face_plane(face):
    geom = getattr(face, "geometry", None)
    if geom is None:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Relation requires face surface geometry",
        )
    origin = getattr(geom, "origin", None)
    normal = getattr(geom, "normal", None)
    if origin is None or normal is None:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Relation requires planar face origin and normal",
        )
    return _vec3(origin), _normalize(_vec3(normal))


def _edge_direction(entity):
    geom = getattr(entity, "geometry", None)
    if geom is None:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Angle and line relations require edge curve geometry",
        )
    if not _is_straight_line_curve(geom):
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Angle and line relations require a positively proven straight-line edge curve; curved or unknown curves cannot be measured exactly from chord endpoints",
            details={"curve_type": _curve_type_of(geom)},
        )
    sp = getattr(geom, "startPoint", None)
    ep = getattr(geom, "endPoint", None)
    if sp is None or ep is None:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Relation requires edge start/end points",
        )
    return _normalize(_sub(_vec3(ep), _vec3(sp)))


def _circular_axis(entity):
    """Derive a concentric axis line (point, direction) from representative Fusion geometry.

    Circular edges expose a circle/arc with center+normal; cylindrical faces expose
    a cylinder with axis origin+axis direction. Radius is not required to establish
    concentricity and is never guessed. Unsupported shapes fail closed.
    """
    kind = _entity_kind(entity)
    geom = getattr(entity, "geometry", None)
    if geom is None:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Concentric relation requires circular geometry",
        )
    if kind in ("edge", "sketch_curve"):
        if not _is_circular_curve(geom):
            raise FusionScriptError(
                "UNSUPPORTED_GEOMETRY",
                "Concentric relation requires a positively proven circular edge (Circle3D/Arc3D); ellipse, elliptical-arc, spline, or unknown curves cannot be concentric circles",
                details={"kind": kind, "curve_type": _curve_type_of(geom)},
            )
        center = getattr(geom, "center", None)
        normal = getattr(geom, "normal", None)
        if center is None or normal is None:
            raise FusionScriptError(
                "UNSUPPORTED_GEOMETRY",
                "Concentric relation requires a circular edge with exact center and normal",
                details={"kind": kind},
            )
        return _vec3(center), _normalize(_vec3(normal)), None
    if kind == "face":
        axis = getattr(geom, "axis", None)
        if axis is None:
            raise FusionScriptError(
                "UNSUPPORTED_GEOMETRY",
                "Concentric relation requires a cylindrical face with an exact axis direction",
                details={"kind": kind},
            )
        # A cylindrical face's axis passes through its axis origin; a circular
        # planar face exposes an explicit center instead.
        origin = getattr(geom, "center", None)
        if origin is None:
            origin = getattr(geom, "origin", None)
        if origin is None:
            raise FusionScriptError(
                "UNSUPPORTED_GEOMETRY",
                "Concentric relation requires a cylindrical face with an exact axis origin",
                details={"kind": kind},
            )
        return _vec3(origin), _normalize(_vec3(axis)), getattr(geom, "radius", None)
    raise FusionScriptError(
        "UNSUPPORTED_GEOMETRY",
        "Concentric relation requires two circular edges or cylindrical faces",
        details={"kind": kind},
    )


def _axis_distance(axis_a, axis_b):
    pa, da, _ra = axis_a
    pb, db, _rb = axis_b
    w = _sub(pa, pb)
    cross_d = _cross(da, db)
    cross_norm = _norm(cross_d)
    if cross_norm < 1e-9:
        cross_w = _cross(w, db)
        denom = _norm(db)
        if denom == 0.0:
            return 0.0
        return _norm(cross_w) / denom
    return abs(_dot(w, cross_d)) / cross_norm


def _measure_angle(entity_a, entity_b):
    kind_a = _entity_kind(entity_a)
    kind_b = _entity_kind(entity_b)
    if kind_a == "face" and kind_b == "face":
        _oa, na = _face_plane(entity_a)
        _ob, nb = _face_plane(entity_b)
        return _angle_deg(na, nb)
    if kind_a in ("edge", "sketch_curve") and kind_b in ("edge", "sketch_curve"):
        da = _edge_direction(entity_a)
        db = _edge_direction(entity_b)
        return _angle_deg(da, db)
    raise FusionScriptError(
        "TYPE_MISMATCH",
        "Angle requires two faces or two edges",
        details={"kind_a": kind_a, "kind_b": kind_b},
    )


def _relation(operation, entity_a, entity_b, ref_a, ref_b, payload):
    kind_a = _entity_kind(entity_a)
    kind_b = _entity_kind(entity_b)

    if operation in ("parallel", "perpendicular"):
        tolerance = float(payload.get("tolerance_deg", 0.01))
        tol = {"value": tolerance, "unit": "deg"}
        if kind_a == "face" and kind_b == "face":
            _oa, na = _face_plane(entity_a)
            _ob, nb = _face_plane(entity_b)
            angle = _angle_deg(na, nb)
        elif kind_a in ("edge", "sketch_curve") and kind_b in ("edge", "sketch_curve"):
            da = _edge_direction(entity_a)
            db = _edge_direction(entity_b)
            angle = _angle_deg(da, db)
        else:
            raise FusionScriptError(
                "TYPE_MISMATCH",
                "Parallel/perpendicular requires two faces or two edges",
                details={"kind_a": kind_a, "kind_b": kind_b},
            )
        parallel_dev = angle if angle <= 90.0 else 180.0 - angle
        if operation == "parallel":
            measured = {"angle_deg": round(parallel_dev, 6)}
            matches = parallel_dev <= tolerance
        else:
            perp_dev = abs(angle - 90.0)
            measured = {"angle_deg": round(perp_dev, 6)}
            matches = perp_dev <= tolerance
        return {
            "relation": operation,
            "matches": matches,
            "measured": measured,
            "tolerance": tol,
            "target_a": ref_a,
            "target_b": ref_b,
        }

    if operation == "coplanar":
        tol_deg = float(payload.get("tolerance_deg", 0.01))
        tol_mm = float(payload.get("tolerance_mm", 0.001))
        tol = {"value": tol_mm, "unit": "mm"}
        if kind_a != "face" or kind_b != "face":
            raise FusionScriptError(
                "TYPE_MISMATCH",
                "Coplanar relation requires two planar faces",
                details={"kind_a": kind_a, "kind_b": kind_b},
            )
        oa, na = _face_plane(entity_a)
        ob, nb = _face_plane(entity_b)
        angle = _angle_deg(na, nb)
        angle_dev = angle if angle <= 90.0 else 180.0 - angle
        distance = _mm(abs(_dot(nb, _sub(ob, oa))))
        matches = angle_dev <= tol_deg and distance <= tol_mm
        return {
            "relation": operation,
            "matches": matches,
            "measured": {
                "angle_deg": round(angle_dev, 6),
                "distance_mm": round(distance, 6),
            },
            "tolerance": tol,
            "tolerances": {"angle_deg": tol_deg, "distance_mm": tol_mm},
            "target_a": ref_a,
            "target_b": ref_b,
        }

    if operation == "concentric":
        tol_deg = float(payload.get("tolerance_deg", 0.01))
        tol_mm = float(payload.get("tolerance_mm", 0.001))
        tol = {"value": tol_mm, "unit": "mm"}
        axis_a = _circular_axis(entity_a)
        axis_b = _circular_axis(entity_b)
        angle = _angle_deg(axis_a[1], axis_b[1])
        angle_dev = angle if angle <= 90.0 else 180.0 - angle
        offset_mm = _mm(_axis_distance(axis_a, axis_b))
        matches = angle_dev <= tol_deg and offset_mm <= tol_mm
        return {
            "relation": operation,
            "matches": matches,
            "measured": {
                "angle_deg": round(angle_dev, 6),
                "offset_mm": round(offset_mm, 6),
            },
            "tolerance": tol,
            "tolerances": {"angle_deg": tol_deg, "offset_mm": tol_mm},
            "target_a": ref_a,
            "target_b": ref_b,
        }

    raise FusionScriptError(
        "UNSUPPORTED_GEOMETRY",
        "Unsupported relation operation",
        details={"operation": operation},
    )


def _face_wall_rect(face, u, v):
    """Return the projected rectangle spec (center_u, center_v, half_u, half_v) of a
    simple wall face, or raise UNSUPPORTED_GEOMETRY.

    Conservative proof: a provable wall face must be a simple quadrilateral with
    exactly one outer loop (no holes) and four distinct corners that project to an
    exact rectangle. Bounding-rectangle overlap alone is NOT conservative because
    concave, holed, or disconnected faces can share a bounding box.
    """
    loops = getattr(face, "loops", None)
    if loops is None or not hasattr(loops, "count") or loops.count != 1:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Face-to-face thickness requires a single outer loop; holed or multi-loop wall faces are ambiguous",
            details={"face_id": getattr(face, "entityToken", None)},
        )
    verts = getattr(face, "vertices", None)
    if verts is None or not hasattr(verts, "count"):
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Face-to-face thickness requires readable face vertices",
            details={"face_id": getattr(face, "entityToken", None)},
        )
    if verts.count != 4:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Face-to-face thickness requires quadrilateral wall faces; concave or multi-vertex faces are ambiguous",
            details={
                "face_id": getattr(face, "entityToken", None),
                "vertex_count": verts.count,
            },
        )
    pts = []
    for i in range(4):
        g = getattr(verts.item(i), "geometry", None)
        if g is None:
            raise FusionScriptError(
                "UNSUPPORTED_GEOMETRY",
                "Face-to-face thickness requires exact vertex geometry",
                details={"face_id": getattr(face, "entityToken", None)},
            )
        gv = _vec3(g)
        pts.append([_dot(gv, u), _dot(gv, v)])
    for i in range(4):
        for j in range(i + 1, 4):
            if abs(pts[i][0] - pts[j][0]) < 1e-9 and abs(pts[i][1] - pts[j][1]) < 1e-9:
                raise FusionScriptError(
                    "UNSUPPORTED_GEOMETRY",
                    "Face-to-face thickness requires four distinct wall-face corners",
                )
    p0, p1, p2, p3 = pts
    e01 = _sub2(p1, p0)
    e12 = _sub2(p2, p1)
    e23 = _sub2(p3, p2)
    e30 = _sub2(p0, p3)
    tol = 1e-6
    if (
        abs(e01[0] + e23[0]) > tol
        or abs(e01[1] + e23[1]) > tol
        or abs(e12[0] + e30[0]) > tol
        or abs(e12[1] + e30[1]) > tol
    ):
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Face-to-face thickness requires a rectangular wall face with parallel opposite edges",
        )
    if abs(e01[0] * e12[0] + e01[1] * e12[1]) > tol:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Face-to-face thickness requires a rectangular wall face with perpendicular adjacent edges",
        )
    xs = [p[0] for p in pts]
    ys = [p[1] for p in pts]
    return (
        (min(xs) + max(xs)) / 2.0,
        (min(ys) + max(ys)) / 2.0,
        (max(xs) - min(xs)) / 2.0,
        (max(ys) - min(ys)) / 2.0,
    )


def _faces_exact_rectangle_coincident(face_a, face_b, normal):
    """Require the two wall faces to project to exactly coincident rectangles.

    Bounding-box overlap alone is not conservative; only exact coincidence of the
    full rectangular footprints proves an unambiguous wall pair.
    """
    n = _normalize(normal)
    u = _perpendicular(n)
    v = _cross(n, u)
    spec_a = _face_wall_rect(face_a, u, v)
    spec_b = _face_wall_rect(face_b, u, v)
    tol = 1e-6  # native cm tolerance for exact correspondence
    for a, b in zip(spec_a, spec_b):
        if abs(a - b) > tol:
            return False
    return True


def _face_to_face_thickness(face_a, face_b, ref_a, ref_b):
    kind_a = _entity_kind(face_a)
    kind_b = _entity_kind(face_b)
    if kind_a != "face" or kind_b != "face":
        raise FusionScriptError(
            "TYPE_MISMATCH",
            "Face-to-face thickness requires two faces",
            details={"kind_a": kind_a, "kind_b": kind_b},
        )
    oa, na = _face_plane(face_a)
    ob, nb = _face_plane(face_b)
    angle = _angle_deg(na, nb)
    # Exact thickness requires anti-parallel (opposite-facing) planar faces.
    if abs(angle - 180.0) > 0.01:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Face-to-face thickness requires parallel opposing planar faces; no whole-body min-wall heuristic",
        )
    # Conservative proof 1: faces must belong to the same solid/body.
    body_a = getattr(face_a, "body", None)
    body_b = getattr(face_b, "body", None)
    if body_a is None or body_b is None or body_a is not body_b:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Face-to-face thickness requires faces on the same solid (provable body relationship)",
        )
    # Positive proof: thickness requires the owning body to be a solid. A
    # surface (non-solid) body has no provable interior material path.
    if getattr(body_a, "isSolid", None) is not True:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Face-to-face thickness requires a solid body (body.isSolid must be True); surface bodies have no provable interior material path",
        )
    # Conservative proof 2: the wall faces must be simple rectangles that coincide
    # exactly in projection. Concave/holed/partial overlap can pass a loose
    # bounding-rectangle test, so only exact coincidence is accepted.
    if not _faces_exact_rectangle_coincident(face_a, face_b, na):
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Face-to-face thickness requires exact coincident rectangular wall faces; concave, holed, or disjoint faces are ambiguous",
        )
    # Conservative proof 3: POSITIVE support envelope. P0 thickness returns an
    # exact value ONLY with positive proof that the same solid is a simple
    # rectangular-prismatic six-face BRep: exactly six faces, every face a simple
    # planar rectangle (no extra faces, no curved/nonplanar faces, no hole or
    # extra/inner-loop topology). Otherwise the material path cannot be proven
    # uninterrupted and the value must fail UNSUPPORTED_GEOMETRY.
    faces = getattr(body_a, "faces", None)
    if faces is None or not hasattr(faces, "count"):
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Face-to-face thickness requires a readable body faces collection; ambiguous topology",
        )
    if faces.count != 6:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Face-to-face thickness requires exactly six planar rectangular faces (rectangular-prismatic solid wall); extra faces are ambiguous",
            details={"face_count": faces.count},
        )
    all_planes = []
    for i in range(faces.count):
        f = faces.item(i)
        if f is face_a or f is face_b:
            continue
        try:
            of, nf = _face_plane(f)
        except FusionScriptError:
            raise FusionScriptError(
                "UNSUPPORTED_GEOMETRY",
                "Face-to-face thickness is ambiguous: a curved/nonplanar face prevents proving a simple uninterrupted uniform solid wall",
                details={"face_id": getattr(f, "entityToken", None)},
            )
        try:
            u = _perpendicular(nf)
            v = _cross(nf, u)
            _face_wall_rect(f, u, v)
        except FusionScriptError:
            raise FusionScriptError(
                "UNSUPPORTED_GEOMETRY",
                "Face-to-face thickness is ambiguous: a non-rectangular or holed/multi-loop face prevents proving a simple uninterrupted uniform solid wall",
                details={"face_id": getattr(f, "entityToken", None)},
            )
        all_planes.append((f, of, nf))
    # Conservative proof 4: an unambiguous wall pair is bounded by exactly two
    # parallel planes. A cavity/hollow part has an internal parallel face strictly
    # between the pair (all faces are now proven planar, so nothing is skipped).
    # The signed bounds are normalized (min/max) so the check is independent of
    # the face_a/face_b order.
    signed_a = _dot(na, oa)
    signed_b = _dot(na, ob)
    lo_signed = min(signed_a, signed_b)
    hi_signed = max(signed_a, signed_b)
    gap_native = hi_signed - lo_signed
    if gap_native <= 1e-9:
        raise FusionScriptError(
            "UNSUPPORTED_GEOMETRY",
            "Face-to-face thickness requires a positive wall gap",
        )
    for f, of, nf in all_planes:
        ang_f = _angle_deg(na, nf)
        if not (ang_f <= 0.01 or abs(ang_f - 180.0) <= 0.01):
            continue
        s_f = _dot(na, of)
        if s_f > lo_signed + 1e-9 and s_f < hi_signed - 1e-9:
            raise FusionScriptError(
                "UNSUPPORTED_GEOMETRY",
                "Face-to-face thickness is ambiguous: an internal parallel face lies between the pair (cavity or partitioned wall)",
                details={"face_id": getattr(f, "entityToken", None)},
            )
    thickness = _mm(abs(_dot(na, _sub(ob, oa))))
    return {
        "quantity": "thickness",
        "value": round(float(thickness), 6),
        "unit": "mm",
        "face_a": ref_a,
        "face_b": ref_b,
        "unambiguous": True,
    }


def run():
    op = PAYLOAD.get("operation", "")
    if op == "test_op":
        return make_result(data={"operation": op}, summary=f"Executed inspect:{op}")

    app, _doc, _design = _get_app_design()

    if op == "describe":
        entity = _resolve_entity(PAYLOAD.get("target"))
        t = _normalize_target(PAYLOAD.get("target")) or {}
        ref = t.get("ref") or "ent_unknown"
        kind = t.get("kind") or _entity_kind(entity)
        name = t.get("name")
        native_type = type(entity).__name__
        data = _describe_entity(entity, ref, kind, name, native_type)
        return make_result(data=data, summary=f"Described {kind} {ref}")

    if op == "bounding_box":
        _world_frame_or_fail(PAYLOAD)
        entity = _resolve_entity(PAYLOAD.get("target"))
        bb = getattr(entity, "boundingBox", None)
        if bb is None:
            raise FusionScriptError(
                "UNSUPPORTED_GEOMETRY",
                "Bounding box unsupported for this entity kind",
            )
        return make_result(
            data={"operation": op, "bounding_box": _bbox_dict(bb)},
            summary="Bounding box measured",
        )

    if op == "oriented_bbox":
        entity = _resolve_entity(PAYLOAD.get("target"))
        return make_result(
            data={"operation": op, "oriented_bbox": _oriented_bbox(entity)},
            summary="Oriented bounding box measured",
        )

    if op == "centroid":
        _world_frame_or_fail(PAYLOAD)
        entity = _resolve_entity(PAYLOAD.get("target"))
        return make_result(
            data={"operation": op, "point": _point_dict(_entity_centroid(entity))},
            summary="Centroid measured",
        )

    if op in ("area", "perimeter", "volume"):
        entity = _resolve_entity(PAYLOAD.get("target"))
        ekind = _entity_kind(entity)
        if op == "area":
            if ekind not in ("body", "face"):
                raise FusionScriptError(
                    "TYPE_MISMATCH",
                    "Area unsupported for this entity kind",
                    details={"kind": ekind},
                )
            value = getattr(entity, "area", None)
            if value is None:
                raise FusionScriptError(
                    "UNSUPPORTED_GEOMETRY",
                    "Area requires exact BRep area property",
                )
            data = {"operation": op, "quantity": "area", "value": float(_mm2(value)), "unit": "mm^2"}
        elif op == "perimeter":
            if ekind == "face":
                value = _face_perimeter(entity)
            elif ekind in ("edge", "sketch_curve"):
                value = getattr(entity, "length", None)
                if value is None:
                    raise FusionScriptError(
                        "UNSUPPORTED_GEOMETRY",
                        "Perimeter requires exact edge length",
                    )
                value = _mm(float(value))
            else:
                raise FusionScriptError(
                    "TYPE_MISMATCH",
                    "Perimeter unsupported for this entity kind",
                    details={"kind": ekind},
                )
            data = {"operation": op, "quantity": "perimeter", "value": float(value), "unit": "mm"}
        else:  # volume
            if ekind != "body":
                raise FusionScriptError(
                    "TYPE_MISMATCH",
                    "Volume unsupported for this entity kind",
                    details={"kind": ekind},
                )
            value = getattr(entity, "volume", None)
            if value is None:
                raise FusionScriptError(
                    "UNSUPPORTED_GEOMETRY",
                    "Volume requires exact BRep volume property",
                )
            data = {"operation": op, "quantity": "volume", "value": float(_mm3(value)), "unit": "mm^3"}
        return make_result(data=data, summary=f"{op} measured")

    if op in ("distance", "minimum_distance"):
        entity_a = _resolve_entity(PAYLOAD.get("target_a"))
        entity_b = _resolve_entity(PAYLOAD.get("target_b"))
        value, from_pt, to_pt = _measure_distance(app, op, entity_a, entity_b)
        return make_result(
            data={
                "operation": op,
                "quantity": op,
                "value": round(float(value), 6),
                "unit": "mm",
                "from": _point_dict(from_pt),
                "to": _point_dict(to_pt),
            },
            summary=f"{op} measured",
        )

    if op == "angle":
        entity_a = _resolve_entity(PAYLOAD.get("target_a"))
        entity_b = _resolve_entity(PAYLOAD.get("target_b"))
        value = _measure_angle(entity_a, entity_b)
        return make_result(
            data={
                "operation": op,
                "quantity": "angle",
                "value": round(float(value), 6),
                "unit": "deg",
            },
            summary="Angle measured",
        )

    if op in ("parallel", "perpendicular", "coplanar", "concentric"):
        entity_a = _resolve_entity(PAYLOAD.get("target_a"))
        entity_b = _resolve_entity(PAYLOAD.get("target_b"))
        ta = _normalize_target(PAYLOAD.get("target_a")) or {}
        tb = _normalize_target(PAYLOAD.get("target_b")) or {}
        ref_a = ta.get("ref") or "ent_a"
        ref_b = tb.get("ref") or "ent_b"
        data = _relation(op, entity_a, entity_b, ref_a, ref_b, PAYLOAD)
        data["operation"] = op
        return make_result(data=data, summary=f"{op} relation measured")

    if op == "face_to_face_thickness":
        entity_a = _resolve_entity(PAYLOAD.get("face_a"))
        entity_b = _resolve_entity(PAYLOAD.get("face_b"))
        ta = _normalize_target(PAYLOAD.get("face_a")) or {}
        tb = _normalize_target(PAYLOAD.get("face_b")) or {}
        ref_a = ta.get("ref") or "ent_a"
        ref_b = tb.get("ref") or "ent_b"
        data = _face_to_face_thickness(entity_a, entity_b, ref_a, ref_b)
        data["operation"] = op
        return make_result(data=data, summary="Face-to-face thickness measured")

    raise FusionScriptError(
        "INVALID_ARGUMENT",
        "Unknown inspection operation",
        details={"operation": op},
    )
