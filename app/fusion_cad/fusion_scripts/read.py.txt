def probe_runtime():
    facts = {}
    app = None
    fusion_version = None
    ui = None
    doc = None
    design = None

    try:
        import adsk.core
        app = adsk.core.Application.get()
        if app:
            facts["has_app"] = True
            fusion_version = getattr(app, "version", None)
            facts["has_measure_manager"] = hasattr(app, "measureManager")
            facts["has_user_interface"] = hasattr(app, "userInterface")
            ui = getattr(app, "userInterface", None)
            if ui:
                facts["has_selection_primitives"] = hasattr(ui, "activeSelections")
                try:
                    selections = getattr(ui, "activeSelections", None)
                    if selections is not None and hasattr(selections, "count"):
                        facts["has_active_selections_context"] = True
                except Exception:
                    pass
            doc = getattr(app, "activeDocument", None)
            if doc:
                facts["has_active_document"] = True
                facts["has_mutation_indicators"] = hasattr(doc, "isModified")
                try:
                    if isinstance(doc.isModified, bool):
                        facts["mutation_indicators_verified"] = True
                except Exception:
                    pass
                facts["has_attributes"] = hasattr(doc, "attributes")
                try:
                    attrs = getattr(doc, "attributes", None)
                    if attrs is not None and hasattr(attrs, "count") and hasattr(attrs, "add"):
                        facts["has_attributes_context"] = True
                except Exception:
                    pass
            viewport = getattr(app, "activeViewport", None)
            if viewport:
                facts["has_active_viewport"] = True
                try:
                    cam = getattr(viewport, "camera", None)
                    if cam and hasattr(cam, "eye") and hasattr(cam, "target") and hasattr(cam, "upVector"):
                        facts["has_active_camera"] = True
                except Exception:
                    pass
                try:
                    if hasattr(viewport, "modelToScreen") and hasattr(viewport, "screenToModel"):
                        facts["has_viewport_conversion_context"] = True
                except Exception:
                    pass
    except Exception as exc:
        facts["adsk_core_probe_error"] = str(exc)

    try:
        import adsk.fusion
        facts["has_adsk_fusion"] = True
        facts["has_design_class"] = hasattr(adsk.fusion, "Design")
        facts["has_sketch_class"] = hasattr(adsk.fusion, "Sketch")
        facts["has_sketch_text"] = hasattr(adsk.fusion, "SketchTexts") or hasattr(adsk.fusion, "SketchText")
        facts["has_export_manager"] = hasattr(adsk.fusion, "ExportManager") or hasattr(adsk.fusion, "DXFExportOptions")
        facts["has_joint_access"] = hasattr(adsk.fusion, "Joints") or hasattr(adsk.fusion, "AsBuiltJoints")

        if app and getattr(app, "activeDocument", None):
            doc = app.activeDocument
            products = getattr(doc, "products", None)
            if products:
                try:
                    design = products.itemByClass("adsk::fusion::Design")
                except Exception:
                    design = None
            if design:
                facts["has_design_access"] = True
                facts["has_timeline_access"] = hasattr(design, "timeline")
                facts["has_entity_token_resolver"] = hasattr(design, "findEntityByToken")
                root = getattr(design, "rootComponent", None)
                facts["has_sketch_access"] = hasattr(root, "sketches") if root else False
            else:
                facts["has_design_access"] = False
                facts["has_timeline_access"] = False
                facts["has_entity_token_resolver"] = False
                facts["has_sketch_access"] = False
        else:
            facts["has_design_access"] = False
            facts["has_timeline_access"] = False
            facts["has_entity_token_resolver"] = False
            facts["has_sketch_access"] = False
    except Exception as exc:
        facts["adsk_fusion_probe_error"] = str(exc)

    try:
        import adsk.core
        facts["has_camera"] = hasattr(adsk.core, "Camera")
        facts["has_viewport_conversion"] = hasattr(adsk.core.Viewport, "modelToScreen") and hasattr(adsk.core.Viewport, "screenToModel")
    except Exception:
        pass

    try:
        import adsk.core
        facts["has_command_preview"] = hasattr(adsk.core, "CommandCreatedEventHandler") and hasattr(adsk.core, "Command")
        facts["has_undo_redo"] = hasattr(adsk.core.Application, "executeTextCommand") or hasattr(adsk.core, "Transaction")
    except Exception:
        pass

    try:
        import adsk.core
        facts["has_attributes"] = facts.get("has_attributes", False) or hasattr(adsk.core, "Attributes")
    except Exception:
        pass

    if "probe_facts" in PAYLOAD and isinstance(PAYLOAD["probe_facts"], dict):
        facts.update(PAYLOAD["probe_facts"])

    probe_err = facts.get("adsk_core_probe_error") or facts.get("adsk_fusion_probe_error")
    probe_failed = bool(probe_err) or (
        not facts.get("has_app")
        and not facts.get("has_adsk_fusion")
        and not facts.get("has_design_access")
    )

    resolved_version = fusion_version or PAYLOAD.get("fusion_version")
    relay_version = PAYLOAD.get("relay_version")

    def err_limits(default_msg):
        limits = [default_msg]
        if facts.get("adsk_core_probe_error"):
            limits.append("adsk.core probe error: " + str(facts["adsk_core_probe_error"]))
        if facts.get("adsk_fusion_probe_error"):
            limits.append("adsk.fusion probe error: " + str(facts["adsk_fusion_probe_error"]))
        return limits

    records = []

    if not probe_failed and facts.get("has_entity_token_resolver") and facts.get("has_design_access"):
        records.append({
            "name": "entity.token_resolver",
            "state": "supported",
            "implementation": "adsk.fusion.Design.findEntityByToken",
            "fusion_version": resolved_version,
            "relay_version": relay_version,
            "limitations": [],
        })
    elif not probe_failed and facts.get("has_entity_token_resolver"):
        records.append({
            "name": "entity.token_resolver",
            "state": "degraded",
            "implementation": "adsk.fusion.Design.findEntityByToken",
            "fusion_version": resolved_version,
            "relay_version": relay_version,
            "limitations": ["Native design entity-token resolver present on class but unverified without active design context"],
        })
    else:
        records.append({
            "name": "entity.token_resolver",
            "state": "unavailable",
            "implementation": None,
            "fusion_version": resolved_version,
            "relay_version": relay_version,
            "limitations": err_limits("Native design entity-token resolver not available on this Fusion runtime"),
        })

    # 2. design.access

            "name": "sketch.access",
            "state": "degraded",
            "implementation": "adsk.fusion.Sketches",
            "fusion_version": resolved_version,
            "relay_version": relay_version,
            "limitations": ["Sketch collection access unverified without active root component context"],
        })
    else:
        records.append({
            "name": "sketch.access",
            "state": "unavailable",
            "implementation": None,
            "fusion_version": resolved_version,
            "relay_version": relay_version,
            "limitations": err_limits("Sketch collection access not available on active root component"),
        })

    # 5. inspect.measure (Finding 3: do not claim contract-level supported from hasattr alone)
    if not probe_failed and facts.get("has_measure_manager"):
        records.append({
            "name": "inspect.measure",
            "state": "degraded",
            "implementation": "adsk.core.MeasureManager",
            "fusion_version": resolved_version,
            "relay_version": relay_version,
            "limitations": ["MeasureManager object presence does not guarantee contract-level geometric measurement without active document and entity context"],
        })
    else:
        records.append({
            "name": "inspect.measure",
            "state": "unavailable",
            "implementation": None,
            "fusion_version": resolved_version,
            "relay_version": relay_version,
            "limitations": err_limits("MeasureManager API not available"),
        })

    # 6. view.camera (Finding 2: property existence is insufficient; prefer degraded/unavailable)
    if not probe_failed and (facts.get("has_camera") or facts.get("has_active_camera") or facts.get("camera_runtime_verified")):
        records.append({
            "name": "view.camera",
            "state": "degraded",
            "implementation": "adsk.core.Camera",
            "fusion_version": resolved_version,
            "relay_version": relay_version,
            "limitations": ["Camera class presence does not guarantee contract-level viewport camera control without active viewport and camera runtime context"],
        })
    else:
        records.append({
            "name": "view.camera",
            "state": "unavailable",
            "implementation": None,
            "fusion_version": resolved_version,
            "relay_version": relay_version,
            "limitations": err_limits("Camera control API not available"),

    # 7. view.viewport_conversion (Finding 2: method presence is insufficient; prefer degraded/unavailable)
    if not probe_failed and (facts.get("has_viewport_conversion") or facts.get("has_viewport_conversion_context") or facts.get("viewport_conversion_verified")):
        records.append({
            "name": "view.viewport_conversion",
            "state": "degraded",
            "implementation": "adsk.core.Viewport",
            "fusion_version": resolved_version,
            "relay_version": relay_version,
            "limitations": ["Viewport screen/model coordinate conversion methods present on class but unverified without active viewport runtime context"],
        })
    else:
        records.append({
            "name": "view.viewport_conversion",
            "state": "unavailable",
            "implementation": None,
            "fusion_version": resolved_version,
            "relay_version": relay_version,
            "limitations": err_limits("Viewport screen/model coordinate conversion methods not available"),
        })

    # 8. view.pick (Finding 3: load-bearing pick remains degraded/unavailable until later live feasibility proof)
    if not probe_failed and facts.get("has_selection_primitives"):
        records.append({
            "name": "view.pick",
            "state": "degraded",
            "implementation": "native-preselect",
            "fusion_version": resolved_version,
            "relay_version": relay_version,
            "limitations": ["Visual pick requires live feasibility proof; load-bearing pick remains degraded pending live verification"],
        })
    elif not probe_failed and facts.get("has_viewport_conversion"):
        records.append({
            "name": "view.pick",
            "state": "degraded",
            "implementation": "viewport-raycast",
            "fusion_version": resolved_version,
            "relay_version": relay_version,
            "limitations": ["Raycast geometry intersection without native preselection; load-bearing pick remains degraded pending live verification"],
        })
    else:
        records.append({
            "name": "view.pick",
            "state": "unavailable",
            "implementation": None,
            "fusion_version": resolved_version,
            "relay_version": relay_version,
            "limitations": err_limits("Neither selection primitives nor viewport projection methods available"),
        })

    # 9. selection.primitives (Finding 2: activeSelections/count is insufficient; prefer degraded/unavailable)
    if not probe_failed and (facts.get("has_selection_primitives") or facts.get("has_active_selections_context") or facts.get("selection_runtime_verified")):
        records.append({
            "name": "selection.primitives",
            "state": "degraded",
            "implementation": "adsk.core.UserInterface.activeSelections",
            "fusion_version": resolved_version,
            "relay_version": relay_version,
            "limitations": ["UserInterface activeSelections attribute present but interactive selection collection runtime behavior unverified without active selection context"],
        })
    else:
        records.append({
            "name": "selection.primitives",
            "state": "unavailable",

            "limitations": err_limits("Interactive selection primitives not available"),
        })

    if not probe_failed and (facts.get("has_command_preview") or facts.get("preview_hooks_verified")):
        records.append({
            "name": "transaction.preview_hooks",
            "state": "degraded",
            "implementation": "adsk.core.Command.executePreview",
            "fusion_version": resolved_version,
            "relay_version": relay_version,
            "limitations": ["Command execution preview hooks present on classes but runtime preview execution unverified without live command lifecycle"],
        })
    else:
        records.append({
            "name": "transaction.preview_hooks",
            "state": "unavailable",
            "implementation": None,
            "fusion_version": resolved_version,
            "relay_version": relay_version,
            "limitations": err_limits("Command execution preview hooks not available"),

    # 11. transaction.preview_replay (Finding 3: load-bearing transaction remains degraded/unavailable until later live feasibility proof)
    if not probe_failed and facts.get("has_command_preview") and facts.get("has_undo_redo"):
        records.append({
            "name": "transaction.preview_replay",
            "state": "degraded",
            "implementation": "command-preview-replay",
            "fusion_version": resolved_version,
            "relay_version": relay_version,
            "limitations": ["Staged transaction preview and replay semantics require live feasibility proof; load-bearing transaction remains degraded pending live verification"],
        })
    elif not probe_failed and facts.get("has_undo_redo"):
        records.append({
            "name": "transaction.preview_replay",
            "state": "degraded",
            "implementation": "undo-redo-fallback",
            "fusion_version": resolved_version,
            "relay_version": relay_version,
            "limitations": ["Command preview hooks not available; rollback relies on active transaction undo; load-bearing transaction remains degraded pending live verification"],
        })
    else:
        records.append({
            "name": "transaction.preview_replay",
            "state": "unavailable",

            "fusion_version": resolved_version,

    if not probe_failed and (facts.get("has_attributes") or facts.get("has_attributes_context") or facts.get("attributes_runtime_verified")):

def run():
    op = PAYLOAD.get("operation", "")
    if op == "capabilities":
        return probe_runtime()
    if op == "echo":
        return make_result(data={"text": PAYLOAD.get("text", "")}, summary="echo")
    if op == "test_op":
        return make_result(data={"operation": op}, summary=f"Executed read:{op}")
    if op == "entity":
        resolved = resolve_fusion_entity(PAYLOAD)
        return make_result(
            data=resolved,
            summary=f"Resolved entity {PAYLOAD.get('ref') or 'entity'}: {resolved.get('outcome')}",
        )

    fp, canonical_data, doc_ref = collect_model_fingerprint(PAYLOAD)
    doc_state = {
        "document_ref": doc_ref,
        "model_revision": PAYLOAD.get("expected_revision") or "rev_1",
        "name": canonical_data.get("document", {}).get("name") or "active",
        "units": "mm",
    }
    result_data = {
        "operation": op,
        "fingerprint": fp,
        "fingerprint_data": canonical_data,
    }
    return make_result(data=result_data, summary=f"Executed read:{op}", document=doc_state)
